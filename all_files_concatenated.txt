use crate::engine::{Engine, StateMachine};
use crate::error::{Error, Result};
use crate::journal::Frames;
pub struct Cursor<'e, S: StateMachine> {
    engine: &'e Engine<S>,
    seq: u64,
    state: S,
    frames: Frames<'e>,
}
impl<'e, S: StateMachine> Cursor<'e, S> {
    pub(crate) fn new(engine: &'e Engine<S>, seq: u64, state: S) -> Self {
        let frames = engine.journal().frames_from(seq + 1);
        Self { engine, seq, state, frames }
    }
    pub fn seq(&self) -> u64 {
        self.seq
    }
    pub fn state(&self) -> &S {
        &self.state
    }
    pub fn into_state(self) -> S {
        self.state
    }
    pub fn at_end(&self) -> bool {
        self.seq == self.engine.last_seq()
    }
    pub fn peek(&self) -> Result<Option<S::Event>> {
        self.event_at(self.seq + 1)
    }
    pub fn step_forward(&mut self) -> Result<Option<S::Event>> {
        let Some(frame) = self.frames.next() else { return Ok(None) };
        debug_assert_eq!(frame.seq, self.seq + 1);
        let event: S::Event = postcard::from_bytes(frame.payload)?;
        self.state.apply(&event);
        self.seq = frame.seq;
        Ok(Some(event))
    }
    pub fn step_back(&mut self) -> Result<Option<S::Event>> {
        if self.seq == 0 {
            return Ok(None);
        }
        let undone = self.seq;
        self.goto(undone - 1)?;
        self.event_at(undone)
    }
    pub fn goto(&mut self, target: u64) -> Result<()> {
        let engine = self.engine;
        let last = engine.last_seq();
        if target > last {
            return Err(Error::SeqOutOfRange { requested: target, last });
        }
        if target < self.seq {
            self.state = engine.state_at(target)?;
            self.seq = target;
            self.frames = engine.journal().frames_from(target + 1);
        } else {
            while self.seq < target {
                if self.step_forward()?.is_none() {
                    break;
                }
            }
        }
        Ok(())
    }
    pub fn run_until(&mut self, mut stop: impl FnMut(&S, u64) -> bool) -> Result<Option<u64>> {
        while self.step_forward()?.is_some() {
            if stop(&self.state, self.seq) {
                return Ok(Some(self.seq));
            }
        }
        Ok(None)
    }
    fn event_at(&self, seq: u64) -> Result<Option<S::Event>> {
        match self.engine.journal().frames_from(seq).next() {
            Some(frame) if frame.seq == seq => Ok(Some(postcard::from_bytes(frame.payload)?)),
            _ => Ok(None),
        }
    }
}

#[derive(Clone, Copy, Default)]
struct Tab {
    n: [usize; 16],
    ok: [usize; 16],
}
impl Tab {
    fn add(&mut self, i: usize, ok: bool) {
        let i = i.min(15);
        self.n[i] += 1;
        self.ok[i] += ok as usize;
    }
    fn show(&self, title: &str, lo: usize, hi: usize) {
        print!("{:<26}", title);
        for i in lo..=hi.min(15) {
            if self.n[i] > 0 {
                print!(" | {}: n={} fid {:.1}%", i, self.n[i], 100.0 * self.ok[i] as f64 / self.n[i] as f64);
            }
        }
        println!();
    }
}
fn pct(a: usize, b: usize) -> f64 {
    100.0 * a as f64 / b.max(1) as f64
}
pub fn run(net: &Net, pr: &Prior, n: usize, seed: u64, horizon: usize) {
    let mut rng = Rng::new(seed ^ 0xD1A6);
    let mut ws = Work::new();
    let tmax = horizon.max(pr.t_hi).max(1);
    let mut hz_n = vec![0usize; tmax];
    let mut hz_rule = vec![0usize; tmax];
    let mut hz_byte = vec![0usize; tmax];
    let (mut by_nt, mut by_m, mut by_k, mut by_pl, mut by_q) =
        (Tab::default(), Tab::default(), Tab::default(), Tab::default(), Tab::default());
    let (mut got, mut n_ok, mut mismatch) = (0usize, 0usize, 0usize);
    let (mut r_tot, mut r_ghost, mut r_rep, mut r_lead) = (0usize, 0usize, 0usize, 0usize);
    let (mut r_near, mut r_conf, mut b_tot, mut b_near) = (0usize, 0usize, 0usize, 0usize);
    let mut b_bits = [0usize; 4];
    let mut guard = 0usize;
    while got < n && guard < 200 * n.max(1) {
        guard += 1;
        let s0 = match draw_world(&mut rng, pr, Part::Held, 1_000_000) {
            Some(s) => s,
            None => continue,
        };
        let s: Sample = if horizon > 0 {
            match rehorizon(&s0, horizon) {
                Some(x) if x.qmax <= pr.q_hi => x,
                _ => continue,
            }
        } else {
            s0
        };
        let st = net.run_sample(&s, 1.0, &mut ws, None);
        let mut failed = false;
        for k in 0..s.t {
            let y = &ws.y[k * NOUT..(k + 1) * NOUT];
            let (b, r) = decode(y);
            let tr = s.trace[k] as usize;
            hz_n[k] += 1;
            if r != tr {
                failed = true;
                hz_rule[k] += 1;
                r_tot += 1;
                if r >= s.prog.k() {
                    r_ghost += 1;
                }
                if k > 0 && r == s.trace[k - 1] as usize {
                    r_rep += 1;
                } else if k + 1 < s.t && r == s.trace[k + 1] as usize {
                    r_lead += 1;
                }
                let margin = y[NB + r] - y[NB + tr];
                if margin < 1.0 {
                    r_near += 1;
                } else if margin > 4.0 {
                    r_conf += 1;
                }
                break;
            }
            if s.flag[k] && b != s.emit[k] {
                failed = true;
                hz_byte[k] += 1;
                b_tot += 1;
                let diff = b ^ s.emit[k];
                b_bits[(diff.count_ones() as usize - 1).min(3)] += 1;
                let mut mn = f64::INFINITY;
                for i in 0..NB {
                    if (diff >> i) & 1 == 1 {
                        mn = mn.min(y[i].abs());
                    }
                }
                if mn < 1.0 {
                    b_near += 1;
                }
                break;
            }
        }
        let ok = !failed;
        if ok {
            n_ok += 1;
        }
        if ok != (st.regen && st.trace_all) {
            mismatch += 1;
        }
        let maxpl = s.prog.prod.iter().map(|w| w.len()).max().unwrap_or(0);
        by_nt.add(nontrivial(&s) as usize, ok);
        by_m.add(s.prog.m, ok);
        by_k.add(s.prog.k(), ok);
        by_pl.add(maxpl, ok);
        by_q.add(s.qmax, ok);
        got += 1;
    }
    println!(
        "=== DIAG : D sur {} mondes non vus (T ∈ [{}, {}], qmax ≤ {}{}) ===",
        got,
        pr.t_lo,
        pr.t_hi,
        pr.q_hi,
        if horizon > 0 { format!(", horizon forcé à {}", horizon) } else { String::new() }
    );
    println!(
        "fid (tous mondes, triviaux compris) : {:.1}%   écarts avec la perte d'entraînement : {} (doit être 0)",
        pct(n_ok, got),
        mismatch
    );
    println!("\nRISQUE PAR TICK : P(première erreur au tick k | tout juste avant), en %");
    println!("{:>4}{:>10}{:>11}{:>11}", "k", "à risque", "règle", "octet");
    for k in 0..tmax {
        if hz_n[k] > 0 {
            println!("{:>4}{:>10}{:>11.2}{:>11.2}", k, hz_n[k], pct(hz_rule[k], hz_n[k]), pct(hz_byte[k], hz_n[k]));
        }
    }
    println!("\nPAR CLASSE (fid exacte)");
    by_nt.show("E constant (0) / non (1)", 0, 1);
    by_m.show("m (pops par tick)", 1, 2);
    by_k.show("k (nombre de règles)", 1, K_MAX);
    by_pl.show("|production| max", 0, W_MAX);
    by_q.show("qmax", 1, 15);
    println!("\nANATOMIE de la première erreur des mondes ratés ({} de règle, {} d'octet)", r_tot, b_tot);
    println!(
        "règle fausse : inexistante {:.0}% | répète le tick précédent {:.0}% | prend celle du tick suivant {:.0}% | marge < 1 (hésitant) {:.0}% | marge > 4 (confiant) {:.0}%",
        pct(r_ghost, r_tot),
        pct(r_rep, r_tot),
        pct(r_lead, r_tot),
        pct(r_near, r_tot),
        pct(r_conf, r_tot)
    );
    println!(
        "octet faux (règle juste) : 1 bit {:.0}% | 2 bits {:.0}% | 3 bits {:.0}% | 4+ bits {:.0}% | bit fautif proche de 0 (|logit| < 1) {:.0}%",
        pct(b_bits[0], b_tot),
        pct(b_bits[1], b_tot),
        pct(b_bits[2], b_tot),
        pct(b_bits[3], b_tot),
        pct(b_near, b_tot)
    );
    println!("lecture : marges < 1 dominantes = bruit d'optimisation ou de précision ; marges > 4 concentrées sur une classe = algorithme absent.");
}

use std::path::{Path, PathBuf};
use serde::{de::DeserializeOwned, Serialize};
use crate::cursor::Cursor;
use crate::error::{Error, Result};
use crate::journal::{Frame, Journal, Recovery};
use crate::snapshot::SnapshotStore;
pub trait StateMachine: Sized + Serialize + DeserializeOwned {
    type Event: Serialize + DeserializeOwned;
    const STATE_VERSION: u32 = 1;
    fn genesis() -> Self;
    fn apply(&mut self, event: &Self::Event);
}
#[derive(Debug, Clone)]
pub struct Options {
    pub snapshot_every: u64,
    pub keep_recent: usize,
    pub milestone_every: u64,
    pub snapshot_dir: Option<PathBuf>,
}
impl Default for Options {
    fn default() -> Self {
        Self { snapshot_every: 10_000, keep_recent: 3, milestone_every: 100_000, snapshot_dir: None }
    }
}
pub struct Engine<S: StateMachine> {
    journal: Journal,
    snapshots: SnapshotStore,
    opts: Options,
    boot_seq: u64,
    state: S,
}
impl<S: StateMachine> Engine<S> {
    pub fn open(path: impl AsRef<Path>) -> Result<(Self, Recovery)> {
        Self::open_with(path, Options::default())
    }
    pub fn open_with(path: impl AsRef<Path>, opts: Options) -> Result<(Self, Recovery)> {
        let path = path.as_ref();
        let (journal, recovery) = Journal::open(path)?;
        let dir = opts.snapshot_dir.clone().unwrap_or_else(|| default_dir(path));
        let mut snapshots = SnapshotStore::open(dir)?;
        snapshots.quarantine_after(journal.last_seq());
        let (mut state, boot_seq) = snapshots.load_best::<S>(&journal, journal.last_seq());
        fold(&mut state, journal.frames_from(boot_seq + 1), u64::MAX)?;
        Ok((Self { journal, snapshots, opts, boot_seq, state }, recovery))
    }
    pub fn submit(&mut self, event: &S::Event) -> Result<u64> {
        let bytes = postcard::to_allocvec(event)?;
        self.submit_encoded(&bytes)
    }
    pub fn submit_encoded(&mut self, bytes: &[u8]) -> Result<u64> {
        let (event, rest) = postcard::take_from_bytes::<S::Event>(bytes)?;
        if !rest.is_empty() {
            return Err(Error::TrailingBytes);
        }
        let seq = self.journal.append(bytes)?;
        self.state.apply(&event);
        Ok(seq)
    }
    pub fn sync(&mut self) -> Result<()> {
        self.journal.sync()
    }
    pub fn maybe_snapshot(&mut self) -> Result<()> {
        let every = self.opts.snapshot_every;
        let since = self.last_seq().saturating_sub(self.snapshots.latest().unwrap_or(0));
        if every != 0 && since >= every {
            self.journal.sync()?; 
            self.write_snapshot()?;
        }
        Ok(())
    }
    pub fn commit(&mut self) -> Result<()> {
        self.sync()?;
        self.maybe_snapshot()
    }
    pub fn snapshot(&mut self) -> Result<u64> {
        self.journal.sync()?; 
        self.write_snapshot()
    }
    fn write_snapshot(&mut self) -> Result<u64> {
        let seq = self.snapshots.save(&self.journal, &self.state)?;
        self.snapshots.prune(self.opts.keep_recent.max(1), self.opts.milestone_every);
        Ok(seq)
    }
    pub fn state(&self) -> &S {
        &self.state
    }
    pub fn last_seq(&self) -> u64 {
        self.journal.last_seq()
    }
    pub fn journal(&self) -> &Journal {
        &self.journal
    }
    pub fn boot_seq(&self) -> u64 {
        self.boot_seq
    }
    pub fn snapshots(&self) -> &[u64] {
        self.snapshots.seqs()
    }
    pub fn state_at(&self, seq: u64) -> Result<S> {
        let last = self.journal.last_seq();
        if seq > last {
            return Err(Error::SeqOutOfRange { requested: seq, last });
        }
        let (mut state, base) = self.snapshots.load_best::<S>(&self.journal, seq);
        fold(&mut state, self.journal.frames_from(base + 1), seq)?;
        Ok(state)
    }
    pub fn cursor(&self) -> Result<Cursor<'_, S>> {
        self.cursor_at(self.last_seq())
    }
    pub fn cursor_at(&self, seq: u64) -> Result<Cursor<'_, S>> {
        Ok(Cursor::new(self, seq, self.state_at(seq)?))
    }
}
fn default_dir(journal: &Path) -> PathBuf {
    let mut name = journal.as_os_str().to_owned();
    name.push(".snapshots");
    PathBuf::from(name)
}
fn fold<'a, S: StateMachine>(
    state: &mut S,
    frames: impl Iterator<Item = Frame<'a>>,
    upto: u64,
) -> Result<()> {
    for frame in frames.take_while(|f| f.seq <= upto) {
        let event: S::Event = postcard::from_bytes(frame.payload)?;
        state.apply(&event);
    }
    Ok(())
}

use std::{fmt, io};
pub type Result<T> = std::result::Result<T, Error>;
#[derive(Debug)]
pub enum Error {
    Io(io::Error),
    Codec(postcard::Error),
    BadHeader(&'static str),
    PayloadTooLarge(usize),
    Locked,
    SeqOutOfRange { requested: u64, last: u64 },
    TrailingBytes,
    Closed,
}
impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Error::Io(e) => write!(f, "i/o error: {e}"),
            Error::Codec(e) => write!(f, "codec error: {e}"),
            Error::BadHeader(why) => write!(f, "invalid journal header: {why}"),
            Error::PayloadTooLarge(n) => write!(f, "payload too large: {n} bytes"),
            Error::Locked => write!(f, "journal is locked by another handle"),
            Error::SeqOutOfRange { requested, last } => {
                write!(f, "cycle {requested} out of range (last = {last})")
            }
            Error::TrailingBytes => write!(f, "trailing bytes after encoded event"),
            Error::Closed => write!(f, "event loop is closed"),
        }
    }
}
impl std::error::Error for Error {
    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
        match self {
            Error::Io(e) => Some(e),
            Error::Codec(e) => Some(e),
            _ => None,
        }
    }
}
impl From<io::Error> for Error {
    fn from(e: io::Error) -> Self {
        Error::Io(e)
    }
}
impl From<postcard::Error> for Error {
    fn from(e: postcard::Error) -> Self {
        Error::Codec(e)
    }
}

pub enum Part {

pub fn prog_hash(p: &Program) -> u64 {
    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
    let mut put = |x: u64| {
        h ^= x;
        h = h.wrapping_mul(0x0000_0100_0000_01b3);
    };
    put(p.m as u64);
    put(0x100 + p.k() as u64);
    for &b in &p.bytes {
        put(b as u64);
    }
    for w in &p.prod {
        put(0x200);
        for &x in w {
            put(x as u64);
        }
    }
    put(0x300);
    for &x in &p.q0 {
        put(x as u64);
    }
    h ^= h >> 32;
    h = h.wrapping_mul(0xd6e8_feb8_6659_fd93);
    h ^ (h >> 32)
}
pub fn in_part(p: &Program, part: Part) -> bool {
    (prog_hash(p) % HELD_MOD == 0) == (part == Part::Held)
}
pub fn draw_world(rng: &mut Rng, pr: &Prior, part: Part, max_att: u32) -> Option<Sample> {
    for _ in 0..512 {
        let s = sample_world(rng, pr, max_att)?;
        if in_part(&s.prog, part) {
            return Some(s);
        }
    }
    None
}
pub fn rehorizon(s: &Sample, t: usize) -> Option<Sample> {
    let r = run(&s.prog, t)?;
    let emit = emit_of(&s.prog, &r.trace);
    let (o, step) = (s.view.o, s.view.s);
    let flag: Vec<bool> = (0..t).map(|i| i >= o && (i - o) % step == 0).collect();
    Some(Sample {
        prog: s.prog.clone(),
        view: s.view,
        t,
        trace: r.trace,
        emit,
        flag,
        qmax: r.qmax,
        bits_p: s.bits_p,
        bits_v: s.bits_v,
        bits_t: s.bits_t,
        attempts: s.attempts,
    })
}

fn rate(a: usize, b: usize) -> f64 {
    a as f64 / b.max(1) as f64
}
impl Agg {
    pub fn add(&mut self, s: &Sample, st: &Stat) {
        self.n += 1;
        self.loss += st.loss;
        self.n_flag += st.n_flag;
        self.flag_ok += st.flag_ok;
        self.n_tick += st.t;
        self.trace_ok += st.trace_ok;
        self.regen += st.regen as usize;
        self.trace_all += st.trace_all as usize;
        self.both += (st.regen && st.trace_all) as usize;
        self.sum_t += s.t;
        self.sum_q += s.qmax;
        self.sum_k += s.prog.k();
        self.diverse += s.emit.iter().any(|&b| b != s.emit[0]) as usize;
    }

        self.n += o.n;
        self.loss += o.loss;

    }
    pub fn rule_acc(&self) -> f64 {
        rate(self.trace_ok, self.n_tick)
    }
    pub fn div_rate(&self) -> f64 {
        rate(self.diverse, self.n)
    }
    pub fn mean_loss(&self) -> f64 {
        self.loss / self.n.max(1) as f64
    }
    pub fn mean_t(&self) -> f64 {
        rate(self.sum_t, self.n)
    }
    pub fn mean_q(&self) -> f64 {
        rate(self.sum_q, self.n)

        rate(self.sum_k, self.n)

        let p = self.fid();

    }

pub fn validate(net: &Net, pr: &Prior, n: usize, rng: &mut Rng, ws: &mut Work) -> Agg {
    let mut a = Agg::default();
    let mut guard = 0usize;
    while a.n < n && guard < 100 * n.max(1) {
        guard += 1;
        if let Some(s) = draw_world(rng, pr, Part::Held, 1_000_000) {
            if !nontrivial(&s) {
                continue;
            }
            let st = net.run_sample(&s, 1.0, ws, None);
            a.add(&s, &st);
        }
    }
    a
}
pub struct ECfg {
    pub n: usize,    
    pub secs: f64,   
    pub seed: u64,
    pub gate: f64,   
}
pub fn axis_len(net: &Net, pr: &Prior, mults: &[usize], ec: &ECfg, ws: &mut Work) -> Vec<Agg> {
    let t1 = pr.t_hi;
    let mut p1 = pr.clone();
    p1.t_lo = t1;
    p1.t_hi = t1;
    let mut rng = Rng::new(ec.seed ^ 0x1E17);
    let mut out = vec![Agg::default(); mults.len()];
    let clock = Instant::now();
    let mut guard = 0usize;
    while out[0].n < ec.n && guard < 400 * ec.n.max(1) && clock.elapsed().as_secs_f64() < ec.secs {
        guard += 1;
        let s = match draw_world(&mut rng, &p1, Part::Held, 100_000) {
            Some(s) => s,
            None => continue,
        };
        if !nontrivial(&s) {
            continue; 
        }
        let mut at: Vec<Sample> = Vec::with_capacity(mults.len());
        for &m in mults {
            match rehorizon(&s, m * t1) {
                Some(x) if x.qmax <= pr.q_hi => at.push(x),
                _ => break,
            }
        }
        if at.len() < mults.len() {
            continue;
        }
        for (i, x) in at.iter().enumerate() {
            let st = net.run_sample(x, 1.0, ws, None);
            out[i].add(x, &st);
        }
    }
    out
}
pub fn axis_depth(net: &Net, pr: &Prior, mults: &[usize], ec: &ECfg, ws: &mut Work) -> Vec<(usize, usize, Agg)> {
    let mut out = Vec::new();
    let mut lo = pr.q_lo;
    for (bi, &m) in mults.iter().enumerate() {
        let hi = m * pr.q_hi;
        let mut pb = pr.clone();
        pb.q_lo = lo;
        pb.q_hi = hi;
        let mut rng = Rng::new(ec.seed ^ (0xDE97 + bi as u64));
        let mut a = Agg::default();
        let clock = Instant::now();
        let mut guard = 0usize;
        while a.n < ec.n && guard < 400 * ec.n.max(1) && clock.elapsed().as_secs_f64() < ec.secs {
            guard += 1;
            if let Some(s) = draw_world(&mut rng, &pb, Part::Held, 100_000) {
                if !nontrivial(&s) {
                    continue;
                }
                let st = net.run_sample(&s, 1.0, ws, None);
                a.add(&s, &st);
            }
        }
        out.push((lo, hi, a));
        lo = hi + 1;
    }
    out
}
pub fn totality(net: &Net, pr: &Prior, ec: &ECfg, ws: &mut Work) -> Agg {
    let mut rng = Rng::new(ec.seed ^ 0xC0FFEE);
    let mut a = Agg::default();

    let mut guard = 0usize;

        guard += 1;
        if let Some(par) = draw_world(&mut rng, pr, Part::Held, 100_000) {

            }

pub fn judge(a1: &Agg, a2: &Agg, gate: f64) -> Verdict {
    let (f1, f2) = (a1.fid(), a2.fid());
    if a1.n < MIN_N || a2.n < MIN_N || f1 < gate {
        Verdict::Inconclusive
    } else if f2 >= 0.8 * f1 {
        Verdict::Pass
    } else if f2 < 0.5 * f1 {
        Verdict::Kill
    } else {
        Verdict::Gray
    }
}
pub fn combine(a: Verdict, b: Verdict) -> Verdict {
    use Verdict::*;
    if a == Kill || b == Kill {
        Kill
    } else if a == Inconclusive || b == Inconclusive {
        Inconclusive
    } else if a == Gray || b == Gray {
        Gray
    } else {
        Pass
    }
}

use std::io;
use std::path::Path;
use memmap2::MmapMut;
use crate::error::{Error, Result};
const MAGIC: [u8; 8] = *b"CHRNLOG\0";
const VERSION: u32 = 1;
const HEADER_LEN: usize = 16;
const FRAME_HEADER: usize = 16;
const MAX_PAYLOAD: usize = 16 << 20;
const MIN_CAPACITY: usize = 4096;
const DEFAULT_CAPACITY: u64 = 1 << 20;
const INDEX_STRIDE: u64 = 1024;
#[derive(Debug, Clone, Copy)]
pub struct Frame<'a> {
    pub seq: u64,
    pub payload: &'a [u8],
}
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub struct Recovery {
    pub frames: u64,
    pub discarded_bytes: u64,
}
pub struct Journal {
    file: File,
    map: MmapMut,
    tail: usize,
    synced: usize,
    next_seq: u64,
    index: Vec<usize>,
}
impl Journal {
    pub fn open(path: impl AsRef<Path>) -> Result<(Self, Recovery)> {
        Self::open_with_capacity(path, DEFAULT_CAPACITY)
    }
    pub fn open_with_capacity(path: impl AsRef<Path>, capacity: u64) -> Result<(Self, Recovery)> {
        let path = path.as_ref();
        let file = OpenOptions::new().read(true).write(true).create(true).open(path)?;
        lock_exclusive(&file)?;
        let existing = file.metadata()?.len();
        if existing == 0 {
            file.set_len(capacity.max(MIN_CAPACITY as u64))?;
        } else if existing < HEADER_LEN as u64 {
            return Err(Error::BadHeader("file shorter than header"));
        }
        let mut map = unsafe { MmapMut::map_mut(&file)? };
        if map[..HEADER_LEN].iter().all(|&b| b == 0) {
            map[0..8].copy_from_slice(&MAGIC);
            map[8..12].copy_from_slice(&VERSION.to_le_bytes());
            map.flush_range(0, HEADER_LEN)?;
            file.sync_all()?;
            sync_parent_dir(path)?;
        } else {
            if map[0..8] != MAGIC[..] {
                return Err(Error::BadHeader("bad magic"));
            }
            let version = u32::from_le_bytes(map[8..12].try_into().unwrap());
            if version != VERSION {
                return Err(Error::BadHeader("unsupported version"));
            }
        }
        let scan = scan_frames(&map);
        let discarded = scrub_tail(&mut map, scan.tail);
        if discarded > 0 {
            map.flush_range(scan.tail, discarded)?;
        }
        let recovery = Recovery { frames: scan.next_seq - 1, discarded_bytes: discarded as u64 };
        let journal = Journal {
            file,
            map,
            tail: scan.tail,
            synced: scan.tail,
            next_seq: scan.next_seq,
            index: scan.index,
        };
        Ok((journal, recovery))
    }
    pub fn append(&mut self, payload: &[u8]) -> Result<u64> {
        if payload.len() > MAX_PAYLOAD {
            return Err(Error::PayloadTooLarge(payload.len()));
        }
        let need = FRAME_HEADER + payload.len();
        if self.tail + need > self.map.len() {
            self.grow(need)?;
        }
        let (off, seq) = (self.tail, self.next_seq);
        let body = off + FRAME_HEADER;
        let len = payload.len() as u32;
        let mut header = [0u8; FRAME_HEADER];
        header[0..4].copy_from_slice(&len.to_le_bytes());
        header[4..8].copy_from_slice(&frame_crc(len, seq, payload).to_le_bytes());
        header[8..16].copy_from_slice(&seq.to_le_bytes());
        self.map[body..body + payload.len()].copy_from_slice(payload);
        self.map[off..body].copy_from_slice(&header);
        if (seq - 1) % INDEX_STRIDE == 0 {
            self.index.push(off);
        }
        self.tail = body + payload.len();
        self.next_seq += 1;
        Ok(seq)
    }
    pub fn sync(&mut self) -> Result<()> {
        if self.tail > self.synced {
            self.map.flush_range(self.synced, self.tail - self.synced)?;
            self.synced = self.tail;
        }
        Ok(())
    }
    pub fn last_seq(&self) -> u64 {
        self.next_seq - 1
    }
    pub fn tail(&self) -> u64 {
        self.tail as u64
    }
    pub fn frames(&self) -> Frames<'_> {
        Frames { buf: &self.map[..self.tail], off: HEADER_LEN }
    }
    pub fn frames_from(&self, seq: u64) -> Frames<'_> {
        let seq = seq.max(1);
        let buf = &self.map[..self.tail];
        if seq >= self.next_seq {
            return Frames { buf, off: buf.len() };
        }
        let mut it = Frames { buf, off: self.index[((seq - 1) / INDEX_STRIDE) as usize] };
        for _ in 0..(seq - 1) % INDEX_STRIDE {
            it.next();
        }
        it
    }
    fn grow(&mut self, need: usize) -> Result<()> {
        let wanted = self.tail + need;
        let mut cap = self.map.len().max(MIN_CAPACITY);
        while cap < wanted {
            cap *= 2;
        }
        self.file.set_len(cap as u64)?;
        self.file.sync_data()?; 
        self.map = unsafe { MmapMut::map_mut(&self.file)? };
        Ok(())
    }
}
pub struct Frames<'a> {
    buf: &'a [u8],
    off: usize,
}
impl<'a> Iterator for Frames<'a> {
    type Item = Frame<'a>;
    fn next(&mut self) -> Option<Frame<'a>> {
        let (len, _crc, seq) = read_header(self.buf, self.off)?;
        let body = self.off + FRAME_HEADER;
        let payload = self.buf.get(body..body.checked_add(len)?)?;
        self.off = body + len;
        Some(Frame { seq, payload })
    }
}
fn read_header(buf: &[u8], off: usize) -> Option<(usize, u32, u64)> {
    let h = buf.get(off..off.checked_add(FRAME_HEADER)?)?;
    let len = u32::from_le_bytes(h[0..4].try_into().ok()?) as usize;
    let crc = u32::from_le_bytes(h[4..8].try_into().ok()?);
    let seq = u64::from_le_bytes(h[8..16].try_into().ok()?);
    Some((len, crc, seq))
}
fn frame_crc(len: u32, seq: u64, payload: &[u8]) -> u32 {
    let mut h = crc32fast::Hasher::new();
    h.update(&len.to_le_bytes());
    h.update(&seq.to_le_bytes());
    h.update(payload);
    h.finalize()
}
struct Scan {
    tail: usize,
    next_seq: u64,
    index: Vec<usize>,
}
fn scan_frames(buf: &[u8]) -> Scan {
    let (mut off, mut next_seq, mut index) = (HEADER_LEN, 1u64, Vec::new());
    while let Some((len, crc, seq)) = read_header(buf, off) {
        if seq != next_seq || len > MAX_PAYLOAD {
            break;
        }
        let body = off + FRAME_HEADER;
        let Some(payload) = buf.get(body..body + len) else { break };
        if frame_crc(len as u32, seq, payload) != crc {
            break;
        }
        if (seq - 1) % INDEX_STRIDE == 0 {
            index.push(off);
        }
        off = body + len;
        next_seq += 1;
    }
    Scan { tail: off, next_seq, index }
}
fn scrub_tail(map: &mut [u8], from: usize) -> usize {
    const CHUNK: usize = 4096;
    let (mut off, mut dirty_end) = (from, from);
    while off < map.len() {
        let end = (off + CHUNK).min(map.len());
        if let Some(last) = map[off..end].iter().rposition(|&b| b != 0) {
            dirty_end = off + last + 1;
            map[off..end].fill(0);
        }
        off = end;
    }
    dirty_end - from
}
#[cfg(unix)]
fn lock_exclusive(file: &File) -> Result<()> {
    use std::os::unix::io::AsRawFd;
    let rc = unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX | libc::LOCK_NB) };
    if rc == 0 {
        return Ok(());
    }
    let err = io::Error::last_os_error();
    Err(if err.kind() == io::ErrorKind::WouldBlock { Error::Locked } else { Error::Io(err) })
}

use std::collections::BTreeMap;
use serde::{Deserialize, Serialize};
use crate::StateMachine;
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum KvEvent {
    Put { key: Vec<u8>, value: Vec<u8> },
    Delete { key: Vec<u8> },
}
#[derive(Debug, Default, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct Kv {
    map: BTreeMap<Vec<u8>, Vec<u8>>,
}
impl Kv {
    pub fn get(&self, key: &[u8]) -> Option<&[u8]> {
        self.map.get(key).map(Vec::as_slice)
    }
    pub fn len(&self) -> usize {
        self.map.len()
    }
    pub fn is_empty(&self) -> bool {
        self.map.is_empty()
    }
    pub fn iter(&self) -> impl Iterator<Item = (&[u8], &[u8])> {
        self.map.iter().map(|(k, v)| (k.as_slice(), v.as_slice()))
    }
}
impl StateMachine for Kv {
    type Event = KvEvent;
    fn genesis() -> Self {
        Self::default()
    }
    fn apply(&mut self, event: &KvEvent) {
        match event {
            KvEvent::Put { key, value } => {
                self.map.insert(key.clone(), value.clone());
            }
            KvEvent::Delete { key } => {
                self.map.remove(key);
            }
        }
    }
}

            })
            .collect();

    }

pub fn sample_program(rng: &mut Rng, pr: &Prior, bits: &mut f64) -> Program {
    let k_cap = pr.k_max.min(K_MAX).min(pr.pool);
    let k = rng.range(1, k_cap);
    *bits += (k_cap as f64).log2();
    let mut pool: Vec<u8> = (0..pr.pool).map(|i| ((b'a' as usize + i) % 256) as u8).collect();
    let mut bytes = Vec::with_capacity(k);
    for _ in 0..k {
        let i = rng.below(pool.len());
        *bits += (pool.len() as f64).log2();
        bytes.push(pool.swap_remove(i));
    }
    let m = if rng.bernoulli(pr.m2) { 2 } else { 1 };
    *bits += if m == 2 { -pr.m2.log2() } else { -(1.0 - pr.m2).log2() };
    let ql_cap = pr.q0_max.min(Q_MAX);
    let ql = rng.range(1, ql_cap);
    *bits += (ql_cap as f64).log2();
    let mut q0 = Vec::with_capacity(ql);
    for _ in 0..ql {
        q0.push(rng.below(k) as u8);
        *bits += (k as f64).log2();
    }
    let wp = if m == 2 { &pr.w2 } else { &pr.w1 };
    let wtot: f64 = wp.iter().sum();
    let mut prod = Vec::with_capacity(k);
    for _ in 0..k {
        let l = rng.categorical(wp);
        *bits += -(wp[l] / wtot).log2();
        let mut w = Vec::with_capacity(l);
        for _ in 0..l {
            w.push(rng.below(k) as u8);
            *bits += (k as f64).log2();
        }
        prod.push(w);
    }
    Program { m, bytes, prod, q0 }
}
pub fn sample_view(rng: &mut Rng, bits: &mut f64) -> View {
    let o = rng.below(4);
    let s = 1 + rng.below(3);
    *bits += 2.0 + 3f64.log2();
    View { o, s }
}
pub struct Run {
    pub trace: Vec<u8>,
    pub qmax: usize,
    pub fired: u32, 
}
pub fn run(p: &Program, t: usize) -> Option<Run> {
    let mut q: VecDeque<u8> = p.q0.iter().copied().collect();
    let mut qmax = q.len();
    let mut trace = Vec::with_capacity(t);
    let mut fired: u32 = 0;
    for _ in 0..t {
        if q.len() < p.m {
            return None;
        }
        let j = q.pop_front()?;
        for _ in 1..p.m {
            let _ = q.pop_front();
        }
        trace.push(j);
        fired |= 1u32 << (j as u32);
        for &x in &p.prod[j as usize] {
            q.push_back(x);
        }
        if q.len() > qmax {
            qmax = q.len();
        }
    }
    Some(Run { trace, qmax, fired })
}
pub fn run_cap(p: &Program, t: usize, cap: usize) -> Option<Run> {
    let mut q: VecDeque<u8> = p.q0.iter().copied().collect();
    let mut qmax = q.len();
    let mut trace = Vec::with_capacity(t);
    let mut fired: u32 = 0;
    for _ in 0..t {
        if q.len() > cap {
            break;
        }
        if q.len() < p.m {
            return None;
        }
        let j = q.pop_front()?;
        for _ in 1..p.m {
            let _ = q.pop_front();
        }
        trace.push(j);
        fired |= 1u32 << (j as u32);
        for &x in &p.prod[j as usize] {
            q.push_back(x);
        }
        if q.len() > qmax {
            qmax = q.len();
        }
    }
    Some(Run { trace, qmax, fired })
}

    let emit = emit_of(&p, &r.trace);
    let flag: Vec<bool> = (0..t).map(|i| i >= view.o && (i - view.o) % view.s == 0).collect();
    Sample { prog: p, view, t, trace: r.trace, emit, flag, qmax: r.qmax, bits_p: bp, bits_v: bv, bits_t: bt, attempts: att }
}
pub fn sample_world(rng: &mut Rng, pr: &Prior, max_attempts: u32) -> Option<Sample> {
    for a in 1..=max_attempts {
        let (mut bp, mut bv) = (0.0f64, 0.0f64);
        let p = sample_program(rng, pr, &mut bp);
        let t = rng.range(pr.t_lo, pr.t_hi);
        let bt = ((pr.t_hi - pr.t_lo + 1) as f64).log2();
        let view = sample_view(rng, &mut bv);
        let r = match run(&p, t) {
            Some(r) => r,
            None => continue,
        };
        if pr.need_all_fired && r.fired != (1u32 << p.k()) - 1 {
            continue;
        }
        if pr.trunc {
            let rc = match run_cap(&p, t, pr.q_hi) {
                Some(rc) => rc,
                None => continue,
            };
            let te = rc.trace.len();
            if te == 0 || te <= view.o {
                continue; 
            }
            return Some(finish(p, view, te, rc, bp, bv, bt, a));
        }
        if r.qmax < pr.q_lo || r.qmax > pr.q_hi {
            continue;
        }
        return Some(finish(p, view, t, r, bp, bv, bt, a));
    }
    None
}
enum Edit {
    Q0Sub(usize, u8),
    ProdSub(usize, usize, u8),
    ProdIns(usize, usize, u8),
    ProdDel(usize, usize),
    FlipM,
}
pub fn mutate(rng: &mut Rng, p: &Program, bits: &mut f64) -> Program {
    let k = p.k();
    let mut ed: Vec<Edit> = Vec::new();
    for i in 0..p.q0.len() {
        for x in 0..k as u8 {
            if x != p.q0[i] {
                ed.push(Edit::Q0Sub(i, x));
            }
        }
    }
    for j in 0..k {
        for i in 0..p.prod[j].len() {
            for x in 0..k as u8 {
                if x != p.prod[j][i] {
                    ed.push(Edit::ProdSub(j, i, x));
                }
            }
            ed.push(Edit::ProdDel(j, i));
        }
        if p.prod[j].len() < W_MAX {
            for i in 0..=p.prod[j].len() {
                for x in 0..k as u8 {
                    ed.push(Edit::ProdIns(j, i, x));
                }
            }
        }
    }
    ed.push(Edit::FlipM);
    *bits += (ed.len() as f64).log2();
    let mut np = p.clone();
    match ed[rng.below(ed.len())] {
        Edit::Q0Sub(i, x) => np.q0[i] = x,
        Edit::ProdSub(j, i, x) => np.prod[j][i] = x,
        Edit::ProdIns(j, i, x) => np.prod[j].insert(i, x),
        Edit::ProdDel(j, i) => {
            np.prod[j].remove(i);
        }
        Edit::FlipM => np.m = 3 - np.m,
    }
    np
}
pub fn mutate_sample(rng: &mut Rng, par: &Sample, pr: &Prior, max_attempts: u32) -> Option<Sample> {
    for a in 1..=max_attempts {
        let mut bp = par.bits_p;
        let np = mutate(rng, &par.prog, &mut bp);
        if pr.trunc {
            if let Some(r) = run_cap(&np, par.t, pr.q_hi) {
                let te = r.trace.len();
                if te > 0 && te > par.view.o {
                    return Some(finish(np, par.view, te, r, bp, par.bits_v, par.bits_t, a));
                }
            }
            continue;
        }
        if let Some(r) = run(&np, par.t) {
            if r.qmax > pr.q_hi {
                continue;
            }
            return Some(finish(np, par.view, par.t, r, bp, par.bits_v, par.bits_t, a));
        }
    }
    None
}
pub fn encode(p: &Program, v: &View, z: &mut [f32]) {
    assert_eq!(z.len(), Z_DIM);
    for x in z.iter_mut() {
        *x = 0.0;
    }
    z[p.m - 1] = 1.0;
    for j in 0..p.k() {
        let base = 2 + j * ROW;
        let b = p.bytes[j];
        for i in 0..8 {
            z[base + i] = if (b >> i) & 1 == 1 { 1.0 } else { -1.0 };
        }
        z[base + 8] = 1.0;
        for s in 0..W_MAX {
            let idx = if s < p.prod[j].len() { p.prod[j][s] as usize } else { NONE };
            z[base + 9 + s * NSLOT + idx] = 1.0;
        }
    }
    let qb = 2 + K_MAX * ROW;
    for s in 0..Q_MAX {
        let idx = if s < p.q0.len() { p.q0[s] as usize } else { NONE };
        z[qb + s * NSLOT + idx] = 1.0;
    }
    z[H_DIM + v.s - 1] = 1.0;
    z[H_DIM + 3 + v.o] = 1.0;
}

pub fn edges() -> Vec<(usize, usize)> {
    let mut e = Vec::new();
    for a in 0..NN {
        for b in a + 1..NN {
            if b - a <= NN / 2 {
                e.push((b, a));
            } else {
                e.push((a, b));
            }
        }
    }
    if SELF_LOOPS {
        for i in 0..NN {
            e.push((i, i));
        }
    }
    e
}
fn check_graph() {
    let mut has = vec![false; NN * NN]; 
    for (d, s) in edges() {
        if d != s {
            assert!(!has[d * NN + s], "arête dupliquée");
            has[d * NN + s] = true;
        }
    }
    for a in 0..NN {
        for b in a + 1..NN {
            assert!(has[b * NN + a] ^ has[a * NN + b], "paire ({a},{b}) : tournoi violé");
        }
    }
    for i in 0..NN {
        assert!(has[((i + 1) % NN) * NN + i], "cycle hamiltonien rompu en {i}");
    }
}

fn bs_mul_add(edges: &[(usize, usize)], blk: &[f64], x: &[f64], out: &mut [f64]) {
    for (e, &(d, s)) in edges.iter().enumerate() {
        let b = &blk[e * BB..(e + 1) * BB];
        let xs = &x[s * ND..(s + 1) * ND];
        let o = &mut out[d * ND..(d + 1) * ND];
        for c in 0..ND {
            let xc = xs[c];
            let col = &b[c * ND..(c + 1) * ND];
            for r in 0..ND {
                o[r] += col[r] * xc;
            }
        }
    }
}
fn bs_mul_t(edges: &[(usize, usize)], blk: &[f64], dd: &[f64], dx: &mut [f64]) {
    for (e, &(d, s)) in edges.iter().enumerate() {
        let b = &blk[e * BB..(e + 1) * BB];
        let dv = &dd[d * ND..(d + 1) * ND];
        let o = &mut dx[s * ND..(s + 1) * ND];
        for c in 0..ND {
            let col = &b[c * ND..(c + 1) * ND];
            let mut acc = 0.0;
            for r in 0..ND {
                acc += col[r] * dv[r];
            }
            o[c] += acc;
        }
    }
}
fn bs_outer(edges: &[(usize, usize)], gblk: &mut [f64], dd: &[f64], x: &[f64]) {
    for (e, &(d, s)) in edges.iter().enumerate() {
        let gb = &mut gblk[e * BB..(e + 1) * BB];
        let dv = &dd[d * ND..(d + 1) * ND];
        let xs = &x[s * ND..(s + 1) * ND];
        for c in 0..ND {
            let xc = xs[c];
            let col = &mut gb[c * ND..(c + 1) * ND];
            for r in 0..ND {
                col[r] += dv[r] * xc;
            }
        }
    }
}
fn cm_mul_add(m: &[f64], x: &[f64], out: &mut [f64]) {
    for c in 0..Z_DIM {
        let xc = x[c];
        if xc == 0.0 {
            continue;
        }
        let col = &m[c * NU..(c + 1) * NU];
        for r in 0..NU {
            out[r] += col[r] * xc;
        }
    }
}
fn cm_outer(gm: &mut [f64], d: &[f64], x: &[f64]) {
    for c in 0..Z_DIM {
        let xc = x[c];
        if xc == 0.0 {
            continue;
        }
        let col = &mut gm[c * NU..(c + 1) * NU];
        for r in 0..NU {
            col[r] += d[r] * xc;
        }
    }
}
pub struct Net {
    pub lay: Lay,
    pub edges: Vec<(usize, usize)>,
    pub p: Vec<f64>,
}
impl Net {
    pub fn new(rng: &mut Rng) -> Net {
        let ed = edges();
        let lay = Lay::new(ed.len());
        let mut indeg = [0usize; NN];
        for &(d, _) in &ed {
            indeg[d] += 1;
        }
        let mut p = vec![0.0f64; lay.total];
        for (e, &(d, _)) in ed.iter().enumerate() {
            let sd = 1.0 / ((ND * indeg[d]) as f64).sqrt();
            for i in 0..BB {
                p[lay.w + e * BB + i] = sd * rng.normal() as f64;
                p[lay.u + e * BB + i] = sd * rng.normal() as f64;
            }
        }
        for i in 0..NU * Z_DIM {
            p[lay.g + i] = 0.15 * rng.normal() as f64;
            p[lay.e + i] = 0.15 * rng.normal() as f64;
        }
        for i in 0..NOUT * NU {
            p[lay.wo + i] = 0.03 * rng.normal() as f64;
            p[lay.vo + i] = 0.03 * rng.normal() as f64;
        }
        Net { lay, edges: ed, p }
    }
    pub fn forward(&self, z32: &[f32], t: usize, ws: &mut Work) {
        let l = &self.lay;
        let ne = l.ne;
        ws.reserve(t);
        for i in 0..Z_DIM {
            ws.z[i] = z32[i] as f64;
        }
        ws.q.copy_from_slice(&self.p[l.gb..l.gb + NU]);
        cm_mul_add(&self.p[l.g..l.g + NU * Z_DIM], &ws.z, &mut ws.q);
        {
            let s0 = &mut ws.s[0..NU];
            s0.copy_from_slice(&self.p[l.e0..l.e0 + NU]);
            cm_mul_add(&self.p[l.e..l.e + NU * Z_DIM], &ws.z, s0);
            for x in s0.iter_mut() {
                *x = x.tanh();
            }
        }
        let wblk = &self.p[l.w..l.w + ne * BB];
        let ublk = &self.p[l.u..l.u + ne * BB];
        for k in 0..t {
            let (sa, sb) = ws.s.split_at_mut((k + 1) * NU);
            let sk = &sa[k * NU..];
            let pk = &mut ws.p[k * NU..(k + 1) * NU];
            pk.copy_from_slice(&self.p[l.ub..l.ub + NU]);
            bs_mul_add(&self.edges, ublk, sk, pk);
            for a in 0..NU {
                ws.b[a] = pk[a] * ws.q[a];
            }
            let yk = &mut ws.y[k * NOUT..(k + 1) * NOUT];
            for o in 0..NOUT {
                let wr = &self.p[l.wo + o * NU..l.wo + (o + 1) * NU];
                let vr = &self.p[l.vo + o * NU..l.vo + (o + 1) * NU];
                let mut acc = self.p[l.bo + o];
                for a in 0..NU {
                    acc += wr[a] * sk[a] + vr[a] * ws.b[a];
                }
                yk[o] = acc;
            }
            if k + 1 < t {
                let sn = &mut sb[0..NU];
                sn.copy_from_slice(&self.p[l.c..l.c + NU]);
                for a in 0..NU {
                    sn[a] += ws.b[a];
                }
                bs_mul_add(&self.edges, wblk, sk, sn);
                for x in sn.iter_mut() {
                    *x = x.tanh();
                }
            }
        }
    }
    pub fn backward(&self, t: usize, ws: &mut Work, g: &mut [f64]) {
        let l = &self.lay;
        let ne = l.ne;
        let wblk = &self.p[l.w..l.w + ne * BB];
        let ublk = &self.p[l.u..l.u + ne * BB];
        for x in ws.ds.iter_mut() {
            *x = 0.0;
        }
        for x in ws.dq.iter_mut() {
            *x = 0.0;
        }
        for k in (0..t).rev() {
            let sk = &ws.s[k * NU..(k + 1) * NU];
            let pk = &ws.p[k * NU..(k + 1) * NU];
            for x in ws.dc.iter_mut() {
                *x = 0.0;
            }
            for x in ws.db.iter_mut() {
                *x = 0.0;
            }
            let dyk = &ws.dy[k * NOUT..(k + 1) * NOUT];
            for o in 0..NOUT {
                let d = dyk[o];
                if d == 0.0 {
                    continue;
                }
                g[l.bo + o] += d;
                let wr = &self.p[l.wo + o * NU..l.wo + (o + 1) * NU];
                let vr = &self.p[l.vo + o * NU..l.vo + (o + 1) * NU];
                for a in 0..NU {
                    let ba = pk[a] * ws.q[a];
                    g[l.wo + o * NU + a] += d * sk[a];
                    g[l.vo + o * NU + a] += d * ba;
                    ws.dc[a] += d * wr[a];
                    ws.db[a] += d * vr[a];
                }
            }
            if k + 1 < t {
                let sn = &ws.s[(k + 1) * NU..(k + 2) * NU];
                for a in 0..NU {
                    let da = ws.ds[a] * (1.0 - sn[a] * sn[a]);
                    ws.da[a] = da;
                    g[l.c + a] += da;
                    ws.db[a] += da;
                }
                bs_outer(&self.edges, &mut g[l.w..l.w + ne * BB], &ws.da, sk);
                bs_mul_t(&self.edges, wblk, &ws.da, &mut ws.dc);
            }
            for a in 0..NU {
                let db = ws.db[a];
                let dp = db * ws.q[a];
                ws.dp[a] = dp;
                ws.dq[a] += db * pk[a];
                g[l.ub + a] += dp;
            }
            bs_outer(&self.edges, &mut g[l.u..l.u + ne * BB], &ws.dp, sk);
            bs_mul_t(&self.edges, ublk, &ws.dp, &mut ws.dc);
            std::mem::swap(&mut ws.ds, &mut ws.dc); 
        }
        let s0 = &ws.s[0..NU];
        for a in 0..NU {
            let da = ws.ds[a] * (1.0 - s0[a] * s0[a]);
            ws.da[a] = da;
            g[l.e0 + a] += da;
        }
        cm_outer(&mut g[l.e..l.e + NU * Z_DIM], &ws.da, &ws.z);
        for a in 0..NU {
            g[l.gb + a] += ws.dq[a];
        }
        cm_outer(&mut g[l.g..l.g + NU * Z_DIM], &ws.dq, &ws.z);
    }
    pub fn run_sample(&self, s: &Sample, w_trace: f64, ws: &mut Work, g: Option<&mut [f64]>) -> Stat {
        let mut z = [0f32; Z_DIM];
        encode(&s.prog, &s.view, &mut z);
        self.forward(&z, s.t, ws);
        let st = loss_raw(s, w_trace, &ws.y, &mut ws.dy);
        if let Some(g) = g {
            self.backward(s.t, ws, g);
        }
        st
    }
    pub fn save(&self, path: &str) -> std::io::Result<()> {
        let mut buf = Vec::with_capacity(12 + self.p.len() * 8);
        buf.extend_from_slice(b"M0D1");
        buf.extend_from_slice(&(self.p.len() as u64).to_le_bytes());
        for x in &self.p {
            buf.extend_from_slice(&x.to_le_bytes());
        }
        std::fs::write(path, buf)
    }
    pub fn load(path: &str) -> std::io::Result<Net> {
        let buf = std::fs::read(path)?;
        let mut net = Net::new(&mut Rng::new(0));
        if buf.len() != 12 + net.p.len() * 8 || buf[..4] != b"M0D1"[..] {
            return Err(std::io::Error::new(std::io::ErrorKind::InvalidData, "fichier modèle invalide"));
        }
        for (i, c) in buf[12..].chunks_exact(8).enumerate() {
            net.p[i] = f64::from_le_bytes(c.try_into().unwrap());
        }
        Ok(net)
    }

pub struct Stat {
    pub t: usize,
    pub loss: f64,      
    pub bits_ce: f64,   
    pub trace_ce: f64,  
    pub n_flag: usize,  
    pub flag_ok: usize, 
    pub trace_ok: usize,
    pub regen: bool,     
    pub trace_all: bool, 
}
fn softplus(x: f64) -> f64 {
    if x > 0.0 { x + (-x).exp().ln_1p() } else { x.exp().ln_1p() }
}
fn sigmoid(x: f64) -> f64 {
    if x >= 0.0 {
        1.0 / (1.0 + (-x).exp())
    } else {
        let e = x.exp();
        e / (1.0 + e)
    }
}
pub fn decode(y: &[f64]) -> (u8, usize) {
    let mut b = 0u8;
    for i in 0..NB {
        if y[i] > 0.0 {
            b |= 1 << i;
        }
    }
    let mut r = 0usize;
    for j in 1..K_MAX {
        if y[NB + j] > y[NB + r] {
            r = j;
        }
    }
    (b, r)
}
pub fn loss_raw(s: &Sample, w_trace: f64, ys: &[f64], dys: &mut [f64]) -> Stat {
    let t = s.t;
    let inv_t = 1.0 / t as f64;
    let mut st = Stat { t, regen: true, trace_all: true, ..Stat::default() };
    let (mut lb, mut lt) = (0.0f64, 0.0f64);
    for k in 0..t {
        let y = &ys[k * NOUT..(k + 1) * NOUT];
        let dy = &mut dys[k * NOUT..(k + 1) * NOUT];
        for d in dy.iter_mut() {
            *d = 0.0;
        }
        if s.flag[k] {
            let byte = s.emit[k];
            let mut l = 0.0;
            for i in 0..NB {
                let tg = ((byte >> i) & 1) as f64;
                l += softplus(y[i]) - tg * y[i];
                dy[i] = (sigmoid(y[i]) - tg) * inv_t / NB as f64;
            }
            lb += l / NB as f64;
            st.n_flag += 1;
            if decode(y).0 == byte {
                st.flag_ok += 1;
            } else {
                st.regen = false;
            }
        }
        let tg = s.trace[k] as usize;
        let lo = &y[NB..];
        let mx = lo.iter().cloned().fold(f64::NEG_INFINITY, f64::max);
        let sum: f64 = lo.iter().map(|v| (v - mx).exp()).sum();
        lt += mx + sum.ln() - lo[tg];
        for j in 0..K_MAX {
            let pj = (lo[j] - mx).exp() / sum;
            dy[NB + j] = w_trace * (pj - if j == tg { 1.0 } else { 0.0 }) * inv_t;
        }
        if decode(y).1 == tg {
            st.trace_ok += 1;
        } else {
            st.trace_all = false;
        }
    }
    st.bits_ce = lb / st.n_flag.max(1) as f64;
    st.trace_ce = lt * inv_t;
    st.loss = (lb + w_trace * lt) * inv_t;
    st
}
pub struct Adam {
    m: Vec<f64>,
    v: Vec<f64>,
    t: i32,
    b1: f64,
    b2: f64,
    eps: f64,
}
impl Adam {
    pub fn new(n: usize) -> Adam {
        Adam { m: vec![0.0; n], v: vec![0.0; n], t: 0, b1: 0.9, b2: 0.999, eps: 1e-8 }
    }
    pub fn step(&mut self, p: &mut [f64], g: &[f64], lr: f64, clip: f64) -> f64 {
        let gn = g.iter().map(|x| x * x).sum::<f64>().sqrt();
        if !gn.is_finite() {
            return gn;
        }
        let sc = if clip > 0.0 && gn > clip { clip / gn } else { 1.0 };
        self.t += 1;
        let c1 = 1.0 - self.b1.powi(self.t);
        let c2 = 1.0 - self.b2.powi(self.t);
        for i in 0..p.len() {
            let gi = g[i] * sc;
            self.m[i] = self.b1 * self.m[i] + (1.0 - self.b1) * gi;
            self.v[i] = self.b2 * self.v[i] + (1.0 - self.b2) * gi * gi;
            p[i] -= lr * (self.m[i] / c1) / ((self.v[i] / c2).sqrt() + self.eps);
        }
        gn
    }
}

use std::future::Future;

use std::sync::{Arc, Condvar, Mutex, MutexGuard, PoisonError};
use std::task::{Context, Poll, Waker};
struct State<T> {
    value: Option<T>,
    waker: Option<Waker>,
    tx_dropped: bool,
    rx_dropped: bool,
}
struct Inner<T> {
    state: Mutex<State<T>>,
    ready: Condvar,
}
fn lock<T>(m: &Mutex<State<T>>) -> MutexGuard<'_, State<T>> {
    m.lock().unwrap_or_else(PoisonError::into_inner)
}
pub(crate) struct Sender<T> {
    inner: Arc<Inner<T>>,
}
pub(crate) struct Receiver<T> {
    inner: Arc<Inner<T>>,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) struct Canceled;
pub(crate) fn channel<T>() -> (Sender<T>, Receiver<T>) {
    let inner = Arc::new(Inner {
        state: Mutex::new(State { value: None, waker: None, tx_dropped: false, rx_dropped: false }),
        ready: Condvar::new(),
    });
    (Sender { inner: inner.clone() }, Receiver { inner })
}
impl<T> Sender<T> {
    pub(crate) fn send(self, value: T) -> Result<(), T> {
        let mut st = lock(&self.inner.state);
        if st.rx_dropped {
            return Err(value);
        }
        st.value = Some(value);
        Ok(())
    }
}
impl<T> Drop for Sender<T> {
    fn drop(&mut self) {
        let waker = {
            let mut st = lock(&self.inner.state);
            st.tx_dropped = true;
            st.waker.take()
        };
        self.inner.ready.notify_all();
        if let Some(waker) = waker {
            waker.wake();
        }
    }
}
impl<T> Receiver<T> {
    pub(crate) fn wait(self) -> Result<T, Canceled> {
        let mut st = lock(&self.inner.state);
        loop {
            if let Some(value) = st.value.take() {
                return Ok(value);
            }
            if st.tx_dropped {
                return Err(Canceled);
            }
            st = self.inner.ready.wait(st).unwrap_or_else(PoisonError::into_inner);
        }
    }
}
impl<T> Drop for Receiver<T> {
    fn drop(&mut self) {
        lock(&self.inner.state).rx_dropped = true;
    }
}
impl<T> Future for Receiver<T> {
    type Output = Result<T, Canceled>;
    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
        let mut st = lock(&self.inner.state);
        if let Some(value) = st.value.take() {
            return Poll::Ready(Ok(value));
        }
        if st.tx_dropped {
            return Poll::Ready(Err(Canceled));
        }
        if !st.waker.as_ref().is_some_and(|w| w.will_wake(cx.waker())) {
            st.waker = Some(cx.waker().clone());
        }
        Poll::Pending
    }
}

#[derive(Clone)]
pub struct Rng {
    s: [u64; 4],
}
fn splitmix64(x: &mut u64) -> u64 {
    *x = x.wrapping_add(0x9E37_79B9_7F4A_7C15);
    let mut z = *x;
    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
    z ^ (z >> 31)
}
impl Rng {
    pub fn new(seed: u64) -> Rng {
        let mut x = seed;
        let s = [
            splitmix64(&mut x),
            splitmix64(&mut x),
            splitmix64(&mut x),
            splitmix64(&mut x),
        ];
        Rng { s }
    }
    pub fn next_u64(&mut self) -> u64 {
        let r = self.s[1].wrapping_mul(5).rotate_left(7).wrapping_mul(9);
        let t = self.s[1] << 17;
        self.s[2] ^= self.s[0];
        self.s[3] ^= self.s[1];
        self.s[1] ^= self.s[2];
        self.s[0] ^= self.s[3];
        self.s[2] ^= t;
        self.s[3] = self.s[3].rotate_left(45);
        r
    }
    pub fn f64(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 * (1.0 / (1u64 << 53) as f64)
    }
    pub fn below(&mut self, n: usize) -> usize {
        (((self.next_u64() >> 32) * (n as u64)) >> 32) as usize
    }
    pub fn range(&mut self, lo: usize, hi: usize) -> usize {
        lo + self.below(hi - lo + 1)
    }
    pub fn bernoulli(&mut self, p: f64) -> bool {
        self.f64() < p
    }
    pub fn normal(&mut self) -> f32 {
        let u1 = self.f64().max(1e-12);
        let u2 = self.f64();
        ((-2.0 * u1.ln()).sqrt() * (2.0 * std::f64::consts::PI * u2).cos()) as f32
    }
    pub fn categorical(&mut self, w: &[f64]) -> usize {
        let tot: f64 = w.iter().sum();
        let mut x = self.f64() * tot;
        for (i, &wi) in w.iter().enumerate() {
            if x < wi {
                return i;
            }
            x -= wi;
        }
        let mut last = 0;
        for (i, &wi) in w.iter().enumerate() {
            if wi > 0.0 {
                last = i;
            }
        }
        last
    }
    pub fn fork(&mut self) -> Rng {
        Rng::new(self.next_u64())
    }
}

            queue: Mutex::new(Queue::new(false)),
            wake: EventFd::new()?,

    fn post<F>(&self, job: F) -> bool
    where
        F: FnOnce(&Ctx<S>) + Send + 'static,
    {
        let rejected = {
            let mut queue = lock(&self.queue);
            if queue.closed {
                Some(job)
            } else {
                queue.jobs.push(Box::new(job));
                None
            }
        };
        if rejected.is_some() {
            return false;
        }
        self.wake.notify();
        true
    }

        let mut queue = lock(&self.queue);
        let closed = queue.closed;
        mem::replace(&mut *queue, Queue::new(closed))
    }

        lock(&self.queue).closed = true;
        self.wake.notify();
    }
    fn close(&self) -> Queue<S> {
        let mut queue = lock(&self.queue);
        mem::replace(&mut *queue, Queue::new(true))
    }

struct Core<S: StateMachine> {
    engine: RefCell<Engine<S>>,
    shared: Arc<Shared<S>>,
    durable: Cell<u64>,
    waiters: RefCell<Vec<Waker>>,
    deferred: RefCell<Vec<Deferred>>,
    spawned: RefCell<Vec<LocalFuture>>,
}
impl<S: StateMachine> Core<S> {
    fn commit_batch(&self) -> Result<()> {
        let last = self.engine.borrow().last_seq();
        let mut outcome: Result<()> = Ok(());
        if last > self.durable.get() {
            outcome = self.engine.borrow_mut().sync();
            if outcome.is_ok() {
                self.durable.set(last);
                self.shared.durable.store(last, Ordering::Release);
                self.shared.syncs.fetch_add(1, Ordering::Relaxed);
            }
        }
        let deferred = mem::take(&mut *self.deferred.borrow_mut());
        let failure = outcome.as_ref().err();
        for done in deferred {
            done(failure);
        }
        if outcome.is_ok() {
            let waiters = mem::take(&mut *self.waiters.borrow_mut());
            for waker in waiters {
                waker.wake();
            }
        }
        outcome
    }
    fn snapshot_if_due(&self, retry_after: &mut u64, backoff: u64) {
        let mut engine = self.engine.borrow_mut();
        let last = engine.last_seq();
        if last >= *retry_after && engine.maybe_snapshot().is_err() {
            self.shared.snapshot_errors.fetch_add(1, Ordering::Relaxed);
            *retry_after = last + backoff;
        }
    }
}

    fn admit(&mut self, ctx: &Ctx<S>) {
        let incoming = mem::take(&mut *ctx.core.spawned.borrow_mut());
        for fut in incoming {
            let id = self.free.pop().unwrap_or(self.slots.len());
            let flag = Arc::new(TaskWaker {
                id,
                queued: AtomicBool::new(false),
                shared: ctx.core.shared.clone(),
            });
            let waker = Waker::from(flag.clone());
            waker.wake_by_ref();
            let task = Some(Task { fut, flag, waker });
            if id == self.slots.len() {
                self.slots.push(task);
            } else {
                self.slots[id] = task;
            }
        }
    }
    fn poll_task(&mut self, id: usize) {
        let Some(Some(task)) = self.slots.get_mut(id) else { return };
        task.flag.queued.store(false, Ordering::Release);
        let mut cx = Context::from_waker(&task.waker);
        if task.fut.as_mut().poll(&mut cx).is_ready() {
            self.slots[id] = None;
            self.free.push(id);
        }
    }

fn reply_when_durable<T: 'static>(
    tx: Sender<Result<T>>,
    value: T,

    move |failure: Option<&Error>| {
        let _ = tx.send(match failure {
            None => Ok(value),
            Some(e) => Err(replicate(e)),
        });
    }
}

    match e {
        Error::Io(e) => Error::Io(io::Error::new(e.kind(), e.to_string())),
        other => Error::Io(io::Error::new(io::ErrorKind::Other, other.to_string())),
    }
}

pub fn block_on<F: Future>(fut: F) -> F::Output {
    let mut fut = std::pin::pin!(fut);
    let waker = Waker::from(Arc::new(ThreadWaker(thread::current())));
    let mut cx = Context::from_waker(&waker);

pub struct Server<S: StateMachine> {
    handle: Handle<S>,
    thread: Option<JoinHandle<Result<()>>>,
}
impl<S: StateMachine + 'static> Server<S> {
    pub fn start(path: impl Into<PathBuf>, opts: Options) -> Result<(Self, Recovery)> {
        let path = path.into();
        let shared = Arc::new(Shared::<S>::new()?);
        let epoll = Epoll::new()?;
        epoll.add(shared.wake.as_raw_fd(), WAKE, EPOLLIN)?;
        let (ready, started) = oneshot::channel::<Result<Recovery>>();
        let thread = {
            let shared = shared.clone();
            thread::Builder::new()
                .name("chronicle-loop".into())
                .spawn(move || run::<S>(path, opts, shared, epoll, ready))?
        };
        match started.wait() {
            Ok(Ok(recovery)) => {
                Ok((Server { handle: Handle { shared }, thread: Some(thread) }, recovery))
            }
            Ok(Err(e)) => {
                let _ = thread.join();
                Err(e)
            }
            Err(_) => match thread.join() {
                Err(payload) => std::panic::resume_unwind(payload),
                Ok(Err(e)) => Err(e),
                Ok(Ok(())) => Err(Error::Closed),
            },
        }
    }
    pub fn handle(&self) -> Handle<S> {
        self.handle.clone()
    }
    pub fn shutdown(mut self) -> Result<()> {
        self.stop()
    }
    fn stop(&mut self) -> Result<()> {
        let Some(thread) = self.thread.take() else { return Ok(()) };
        self.handle.shared.request_close();
        match thread.join() {
            Ok(result) => result,
            Err(payload) => std::panic::resume_unwind(payload),
        }
    }
}
impl<S: StateMachine> Drop for Server<S> {
    fn drop(&mut self) {
        if let Some(thread) = self.thread.take() {
            self.handle.shared.request_close();
            let _ = thread.join(); 
        }
    }
}
fn run<S: StateMachine + 'static>(
    path: PathBuf,
    opts: Options,
    shared: Arc<Shared<S>>,
    epoll: Epoll,
    ready: Sender<Result<Recovery>>,
) -> Result<()> {
    let backoff = opts.snapshot_every.max(1);
    let (engine, recovery) = match Engine::<S>::open_with(&path, opts) {
        Ok(opened) => opened,
        Err(e) => {
            let _ = ready.send(Err(e));
            return Ok(());
        }
    };
    let _closer = CloseOnExit(shared.clone());
    let last = engine.last_seq();
    shared.durable.store(last, Ordering::Release);
    let ctx = Ctx {
        core: Rc::new(Core {
            engine: RefCell::new(engine),
            shared: shared.clone(),
            durable: Cell::new(last),
            waiters: RefCell::new(Vec::new()),
            deferred: RefCell::new(Vec::new()),
            spawned: RefCell::new(Vec::new()),
        }),
    };
    let _ = ready.send(Ok(recovery));
    let mut tasks = Executor::<S>::new();
    let mut tokens = Vec::new();
    let mut retry_after = 0;
    loop {
        tokens.clear();
        epoll.wait(&mut tokens, None)?;
        shared.wake.drain();
        let batch = shared.take();
        for job in batch.jobs {
            job(&ctx);
        }
        tasks.admit(&ctx);
        for id in batch.woken {
            tasks.poll_task(id);
        }
        tasks.admit(&ctx);
        ctx.core.commit_batch()?;
        ctx.core.snapshot_if_due(&mut retry_after, backoff);
        if batch.closed {
            let mut engine = ctx.core.engine.borrow_mut();
            if engine.snapshots().last() != Some(&engine.last_seq()) {
                let _ = engine.snapshot();
            }
            return Ok(());
        }
    }
}

pub fn queue_lens(p: &Program, t: usize) -> Vec<usize> {
    let mut q: VecDeque<u8> = p.q0.iter().copied().collect();
    let mut out = Vec::with_capacity(t);
    for _ in 0..t {
        out.push(q.len());
        let j = match q.pop_front() {
            Some(j) => j as usize,
            None => break,
        };
        for _ in 1..p.m {
            let _ = q.pop_front();
        }
        for &x in &p.prod[j] {
            q.push_back(x);
        }
    }
    out
}

const MAGIC: [u8; 8] = *b"CHRNSNP\0";
const FORMAT: u32 = 1;
const HEADER: usize = 28;
pub(crate) struct SnapshotStore {
    dir: PathBuf,
    seqs: Vec<u64>,
}
impl SnapshotStore {
    pub(crate) fn open(dir: PathBuf) -> Result<Self> {
        let mut seqs: Vec<u64> = Vec::new();
        match fs::read_dir(&dir) {
            Ok(entries) => {
                for entry in entries {
                    let path = entry?.path();
                    match path.extension().and_then(|e| e.to_str()) {
                        Some("tmp") => {
                            let _ = fs::remove_file(&path);
                        }
                        Some("snap") => {
                            let stem = path.file_stem().and_then(|s| s.to_str());
                            if let Some(seq) = stem.and_then(|s| s.parse::<u64>().ok()) {
                                seqs.push(seq);
                            }
                        }
                        _ => {}
                    }
                }
            }
            Err(e) if e.kind() == io::ErrorKind::NotFound => {}
            Err(e) => return Err(e.into()),
        }
        seqs.sort_unstable();
        Ok(Self { dir, seqs })
    }
    pub(crate) fn seqs(&self) -> &[u64] {
        &self.seqs
    }
    pub(crate) fn latest(&self) -> Option<u64> {
        self.seqs.last().copied()
    }
    pub(crate) fn save<S: StateMachine>(&mut self, journal: &Journal, state: &S) -> Result<u64> {
        let seq = journal.last_seq();
        if seq == 0 {
            return Ok(0);
        }
        let payload = postcard::to_allocvec(state)?;
        let mut head = [0u8; HEADER];
        head[0..8].copy_from_slice(&MAGIC);
        head[8..12].copy_from_slice(&FORMAT.to_le_bytes());
        head[12..16].copy_from_slice(&S::STATE_VERSION.to_le_bytes());
        head[16..24].copy_from_slice(&seq.to_le_bytes());
        head[24..28].copy_from_slice(&anchor_of(journal, seq).unwrap_or_default().to_le_bytes());
        let mut crc = crc32fast::Hasher::new();
        crc.update(&head);
        crc.update(&payload);
        let trailer = crc.finalize().to_le_bytes();
        fs::create_dir_all(&self.dir)?;
        let parts: [&[u8]; 3] = [&head, &payload, &trailer];
        write_atomic(&self.dir, &snap_name(seq), &parts)?;
        if let Err(pos) = self.seqs.binary_search(&seq) {
            self.seqs.insert(pos, seq);
        }
        Ok(seq)
    }
    pub(crate) fn load_best<S: StateMachine>(&self, journal: &Journal, upto: u64) -> (S, u64) {
        for &seq in self.seqs.iter().rev() {
            if seq > upto {
                continue;
            }
            if let Some(state) = self.load::<S>(journal, seq) {
                return (state, seq);
            }
        }
        (S::genesis(), 0)
    }
    fn load<S: StateMachine>(&self, journal: &Journal, seq: u64) -> Option<S> {
        let bytes = fs::read(snap_path(&self.dir, seq)).ok()?;
        let (body, trailer) = bytes.split_at(bytes.len().checked_sub(4)?);
        if body.len() < HEADER || crc32fast::hash(body) != u32::from_le_bytes(trailer.try_into().ok()?) {
            return None;
        }
        let u32_at = |o: usize| u32::from_le_bytes(body[o..o + 4].try_into().unwrap());
        let u64_at = |o: usize| u64::from_le_bytes(body[o..o + 8].try_into().unwrap());
        let valid = body[0..8] == MAGIC[..]
            && u32_at(8) == FORMAT
            && u32_at(12) == S::STATE_VERSION
            && u64_at(16) == seq
            && Some(u32_at(24)) == anchor_of(journal, seq);
        if !valid {
            return None;
        }
        let (state, rest) = postcard::take_from_bytes::<S>(&body[HEADER..]).ok()?;
        rest.is_empty().then_some(state)
    }
    pub(crate) fn quarantine_after(&mut self, seq: u64) {
        let keep = self.seqs.partition_point(|&s| s <= seq);
        for s in self.seqs.drain(keep..) {
            let path = snap_path(&self.dir, s);
            let _ = fs::rename(&path, path.with_extension("orphan"));
        }
    }
    pub(crate) fn prune(&mut self, keep_recent: usize, milestone_every: u64) {
        let recent_from = self.seqs.len().saturating_sub(keep_recent);
        let mut previous = None;
        let mut kept = Vec::with_capacity(self.seqs.len());
        for (i, &seq) in self.seqs.iter().enumerate() {
            let bucket = (milestone_every != 0).then(|| seq / milestone_every);
            let milestone = bucket.is_some() && bucket != previous;
            previous = bucket;
            if i >= recent_from || milestone {
                kept.push(seq);
            } else {
                let _ = fs::remove_file(snap_path(&self.dir, seq));
            }
        }
        self.seqs = kept;
    }
}

fn anchor_of(journal: &Journal, seq: u64) -> Option<u32> {
    let frame = journal.frames_from(seq).next().filter(|f| f.seq == seq)?;
    let mut h = crc32fast::Hasher::new();
    h.update(&frame.seq.to_le_bytes());
    h.update(frame.payload);
    Some(h.finalize())
}
fn write_atomic(dir: &Path, name: &str, parts: &[&[u8]]) -> io::Result<()> {
    let tmp = dir.join(format!("{name}.tmp"));
    let written = (|| -> io::Result<()> {
        let mut file = File::create(&tmp)?;
        for part in parts {
            file.write_all(part)?;
        }
        file.sync_all()?;
        fs::rename(&tmp, dir.join(name))
    })();
    if written.is_err() {
        let _ = fs::remove_file(&tmp);
        return written;
    }
    sync_dir(dir)
}

fn cvt(rc: libc::c_int) -> io::Result<libc::c_int> {
    if rc < 0 {
        Err(io::Error::last_os_error())
    } else {
        Ok(rc)
    }
}
pub(crate) struct EventFd(OwnedFd);
impl EventFd {
    pub(crate) fn new() -> io::Result<Self> {
        let raw = cvt(unsafe { libc::eventfd(0, libc::EFD_NONBLOCK | libc::EFD_CLOEXEC) })?;
        Ok(Self(unsafe { OwnedFd::from_raw_fd(raw) }))
    }
    pub(crate) fn notify(&self) {
        let one = 1u64.to_ne_bytes();
        loop {
            let n = unsafe { libc::write(self.0.as_raw_fd(), one.as_ptr().cast(), one.len()) };
            if n >= 0 || io::Error::last_os_error().kind() != io::ErrorKind::Interrupted {
                return;
            }
        }
    }
    pub(crate) fn drain(&self) {
        let mut buf = [0u8; 8];
        unsafe { libc::read(self.0.as_raw_fd(), buf.as_mut_ptr().cast(), buf.len()) };
    }
}
impl AsRawFd for EventFd {
    fn as_raw_fd(&self) -> RawFd {
        self.0.as_raw_fd()
    }
}
pub(crate) struct Epoll(OwnedFd);
impl Epoll {
    pub(crate) fn new() -> io::Result<Self> {
        let raw = cvt(unsafe { libc::epoll_create1(libc::EPOLL_CLOEXEC) })?;
        Ok(Self(unsafe { OwnedFd::from_raw_fd(raw) }))
    }
    pub(crate) fn add(&self, fd: RawFd, token: u64, events: u32) -> io::Result<()> {
        let mut event = libc::epoll_event { events, u64: token };
        cvt(unsafe { libc::epoll_ctl(self.0.as_raw_fd(), libc::EPOLL_CTL_ADD, fd, &mut event) })?;
        Ok(())
    }
    pub(crate) fn wait(&self, out: &mut Vec<u64>, timeout: Option<Duration>) -> io::Result<()> {
        const MAX: usize = 64;
        let millis = match timeout {
            None => -1,
            Some(d) => d.as_millis().min(i32::MAX as u128) as libc::c_int,
        };
        let mut events: Vec<libc::epoll_event> = Vec::with_capacity(MAX);
        let n = unsafe {
            libc::epoll_wait(self.0.as_raw_fd(), events.as_mut_ptr(), MAX as libc::c_int, millis)
        };
        if n < 0 {
            let err = io::Error::last_os_error();
            return if err.kind() == io::ErrorKind::Interrupted { Ok(()) } else { Err(err) };
        }
        unsafe { events.set_len(n as usize) };
        out.extend(events.iter().map(|e| e.u64));
        Ok(())
    }
}

    } else {

impl Net {
    pub fn new(rng: &mut Rng) -> Net {
        let lay = Lay::new();
        let l = lay;
        let mut p = vec![0.0f64; lay.total];
        fill(rng, &mut p[l.l1..l.l1 + DC * NSLOT], 0.5);
        fill(rng, &mut p[l.l2..l.l2 + DC * NSLOT], 0.5);
        fill(rng, &mut p[l.a..l.a + NO * DC * DC], 0.15);
        for d in 0..DC {
            p[l.a + d * DC + d] += 1.0; 
            p[l.gam + d] = 2.0; 
        }
        for i in DC..NO * DC {
            p[l.gam + i] = 0.3;
        }
        fill(rng, &mut p[l.gg..l.gg + NO * DC * DW], 0.05);
        fill(rng, &mut p[l.ua..l.ua + DC], 0.5);
        fill(rng, &mut p[l.uv..l.uv + DR * DC], 0.25);
        fill(rng, &mut p[l.vw..l.vw + DW * DR], 0.25);
        for i in 0..DW {
            p[l.vb + i] = 0.5;
        }
        fill(rng, &mut p[l.gw..l.gw + DW * Z_DIM], 0.15);
        fill(rng, &mut p[l.wo..l.wo + NOUT * DR], 0.03);
        fill(rng, &mut p[l.vo..l.vo + NOUT * DW], 0.03);
        Net { lay, p, rho: f64::INFINITY }
    }
    pub fn forward(&self, z32: &[f32], t: usize, ws: &mut Work) {
        let (l, p) = (&self.lay, &self.p);
        let n = cells(t);
        ws.reserve(t, n);
        ws.n = n;
        for i in 0..Z_DIM {
            ws.z[i] = z32[i] as f64;
        }
        for u in 0..DW {
            let row = &p[l.gw + u * Z_DIM..l.gw + (u + 1) * Z_DIM];
            let mut a = p[l.gb + u];
            for c in 0..Z_DIM {
                a += row[c] * ws.z[c];
            }
            ws.qz[u] = a;
        }
        for i in 0..n {
            let xi = xs(&ws.z, i);
            let xp = if i > 0 { xs(&ws.z, i - 1) } else { [0.0; NSLOT] };
            for d in 0..DC {
                let mut a = p[l.l0 + d];
                for c in 0..NSLOT {
                    a += p[l.l1 + d * NSLOT + c] * xi[c] + p[l.l2 + d * NSLOT + c] * xp[c];
                }
                ws.s[i * DC + d] = a.tanh();
            }
        }
        for k in 0..t {
            let na = n.min(3 * k + 1);
            let (sa, sb) = ws.s.split_at_mut((k + 1) * n * DC);
            let sk = &sa[k * n * DC..];
            let mut mx = f64::NEG_INFINITY;
            for i in 0..na {
                let mut lg = 0.0;
                for d in 0..DC {
                    lg += p[l.ua + d] * sk[i * DC + d];
                }
                ws.att[k * n + i] = lg;
                if lg > mx {
                    mx = lg;
                }
            }
            let mut sum = 0.0;
            for i in 0..na {
                let e = (ws.att[k * n + i] - mx).exp();
                ws.att[k * n + i] = e;
                sum += e;
            }
            let rk = &mut ws.r[k * DR..(k + 1) * DR];
            for x in rk.iter_mut() {
                *x = 0.0;
            }
            for i in 0..na {
                let a = ws.att[k * n + i] / sum;
                ws.att[k * n + i] = a;
                for j in 0..DR {
                    let mut vj = p[l.uvb + j];
                    for d in 0..DC {
                        vj += p[l.uv + j * DC + d] * sk[i * DC + d];
                    }
                    ws.v[(k * n + i) * DR + j] = vj;
                    rk[j] += a * vj;
                }
            }
            for u in 0..DW {
                let mut a = p[l.vb + u];
                for j in 0..DR {
                    a += p[l.vw + u * DR + j] * rk[j];
                }
                ws.pw[k * DW + u] = a;
                ws.w[k * DW + u] = a * ws.qz[u];
            }
            for o in 0..NO {
                for d in 0..DC {
                    let row = &p[l.gg + (o * DC + d) * DW..l.gg + (o * DC + d + 1) * DW];
                    let mut g = p[l.gam + o * DC + d];
                    for u in 0..DW {
                        g += row[u] * ws.w[k * DW + u];
                    }
                    ws.gt[(k * NO + o) * DC + d] = g;
                }
            }
            for o in 0..NOUT {
                let mut a = p[l.bo + o];
                for j in 0..DR {
                    a += p[l.wo + o * DR + j] * rk[j];
                }
                for u in 0..DW {
                    a += p[l.vo + o * DW + u] * ws.w[k * DW + u];
                }
                ws.y[k * NOUT + o] = a;
            }
            if k + 1 < t {
                let sn = &mut sb[0..n * DC];
                for i in 0..n {
                    for d in 0..DC {
                        let mut a = p[l.bc + d];
                        for o in 0..NO.min(i + 1) {
                            let src = &sk[(i - o) * DC..(i - o + 1) * DC];
                            let row = &p[l.a + (o * DC + d) * DC..l.a + (o * DC + d + 1) * DC];
                            let mut u = 0.0;
                            for c in 0..DC {
                                u += row[c] * src[c];
                            }
                            a += ws.gt[(k * NO + o) * DC + d] * u;
                        }
                        sn[i * DC + d] = a.tanh();
                    }
                }
            }
        }
    }
    pub fn backward(&self, t: usize, ws: &mut Work, g: &mut [f64]) {
        let (l, p) = (&self.lay, &self.p);
        let n = ws.n;
        let mut dqz = [0.0f64; DW];
        let mut dw = [0.0f64; DW];
        let mut dr = [0.0f64; DR];
        let mut dg = [0.0f64; NO * DC];
        for x in ws.ds[..n * DC].iter_mut() {
            *x = 0.0;
        }
        for k in (0..t).rev() {
            let na = n.min(3 * k + 1);
            for x in dg.iter_mut() {
                *x = 0.0;
            }
            for x in dw.iter_mut() {
                *x = 0.0;
            }
            for x in dr.iter_mut() {
                *x = 0.0;
            }
            if k + 1 < t {
                let sk = &ws.s[k * n * DC..(k + 1) * n * DC];
                let sn = &ws.s[(k + 1) * n * DC..(k + 2) * n * DC];
                for i in 0..n {
                    for d in 0..DC {
                        let sv = sn[i * DC + d];
                        let da = ws.dsn[i * DC + d] * (1.0 - sv * sv);
                        if da == 0.0 {
                            continue;
                        }
                        g[l.bc + d] += da;
                        for o in 0..NO.min(i + 1) {
                            let src = &sk[(i - o) * DC..(i - o + 1) * DC];
                            let ro = l.a + (o * DC + d) * DC;
                            let mut u = 0.0;
                            for c in 0..DC {
                                u += p[ro + c] * src[c];
                            }
                            dg[o * DC + d] += da * u;
                            let du = da * ws.gt[(k * NO + o) * DC + d];
                            for c in 0..DC {
                                g[ro + c] += du * src[c];
                                ws.ds[(i - o) * DC + c] += p[ro + c] * du;
                            }
                        }
                    }
                }
            }
            if k + 1 < t && self.rho.is_finite() {
                let nn = n * DC;
                let nr = ws.dsn[..nn].iter().map(|x| x * x).sum::<f64>().sqrt();
                let nc = ws.ds[..nn].iter().map(|x| x * x).sum::<f64>().sqrt();
                ws.ticks += 1;
                if nc > self.rho * nr {
                    let sc = self.rho * nr / nc;
                    for x in ws.ds[..nn].iter_mut() {
                        *x *= sc;
                    }
                    ws.capped += 1;
                }
            }
            for o in 0..NO {
                for d in 0..DC {
                    let v = dg[o * DC + d];
                    if v == 0.0 {
                        continue;
                    }
                    g[l.gam + o * DC + d] += v;
                    let ro = l.gg + (o * DC + d) * DW;
                    for u in 0..DW {
                        g[ro + u] += v * ws.w[k * DW + u];
                        dw[u] += p[ro + u] * v;
                    }
                }
            }
            for o in 0..NOUT {
                let d = ws.dy[k * NOUT + o];
                if d == 0.0 {
                    continue;
                }
                g[l.bo + o] += d;
                for j in 0..DR {
                    g[l.wo + o * DR + j] += d * ws.r[k * DR + j];
                    dr[j] += p[l.wo + o * DR + j] * d;
                }
                for u in 0..DW {
                    g[l.vo + o * DW + u] += d * ws.w[k * DW + u];
                    dw[u] += p[l.vo + o * DW + u] * d;
                }
            }
            for u in 0..DW {
                let dwu = dw[u];
                if dwu == 0.0 {
                    continue;
                }
                let dpw = dwu * ws.qz[u];
                dqz[u] += dwu * ws.pw[k * DW + u];
                g[l.vb + u] += dpw;
                for j in 0..DR {
                    g[l.vw + u * DR + j] += dpw * ws.r[k * DR + j];
                    dr[j] += p[l.vw + u * DR + j] * dpw;
                }
            }
            let mut ssum = 0.0;
            for i in 0..na {
                let mut da = 0.0;
                for j in 0..DR {
                    da += dr[j] * ws.v[(k * n + i) * DR + j];
                }
                ws.dl[i] = da;
                ssum += ws.att[k * n + i] * da;
            }
            for i in 0..na {
                let a = ws.att[k * n + i];
                let dl = a * (ws.dl[i] - ssum);
                let si = &ws.s[(k * n + i) * DC..(k * n + i + 1) * DC];
                for j in 0..DR {
                    let dv = a * dr[j];
                    g[l.uvb + j] += dv;
                    for d in 0..DC {
                        g[l.uv + j * DC + d] += dv * si[d];
                        ws.ds[i * DC + d] += p[l.uv + j * DC + d] * dv;
                    }
                }
                for d in 0..DC {
                    g[l.ua + d] += dl * si[d];
                    ws.ds[i * DC + d] += dl * p[l.ua + d];
                }
            }
            std::mem::swap(&mut ws.ds, &mut ws.dsn);
            for x in ws.ds[..n * DC].iter_mut() {
                *x = 0.0;
            }
        }
        for i in 0..n {
            let xi = xs(&ws.z, i);
            let xp = if i > 0 { xs(&ws.z, i - 1) } else { [0.0; NSLOT] };
            for d in 0..DC {
                let s0 = ws.s[i * DC + d];
                let da = ws.dsn[i * DC + d] * (1.0 - s0 * s0);
                g[l.l0 + d] += da;
                for c in 0..NSLOT {
                    g[l.l1 + d * NSLOT + c] += da * xi[c];
                    g[l.l2 + d * NSLOT + c] += da * xp[c];
                }
            }
        }
        for u in 0..DW {
            g[l.gb + u] += dqz[u];
            if dqz[u] == 0.0 {
                continue;
            }
            for c in 0..Z_DIM {
                g[l.gw + u * Z_DIM + c] += dqz[u] * ws.z[c];
            }
        }
    }
    pub fn run_sample(&self, s: &Sample, w_trace: f64, ws: &mut Work, g: Option<&mut [f64]>) -> Stat {
        let mut z = [0f32; Z_DIM];
        encode(&s.prog, &s.view, &mut z);
        self.forward(&z, s.t, ws);
        let st = loss_raw(s, w_trace, &ws.y, &mut ws.dy);
        if let Some(g) = g {
            self.backward(s.t, ws, g);
        }
        st
    }

pub fn gradcheck(seed: u64) -> bool {
    let pr = Prior::demo();
    let mut rng = Rng::new(seed);
    let mut net = Net::new(&mut rng);
    let lay = net.lay;
    assert_eq!(lay.groups().iter().map(|g| g.2).sum::<usize>(), lay.total);
    println!(
        "D (bande) : {} paramètres ; cellule {} unités, bus de lecture {}, bus d'écriture {}",
        net.p.len(),
        DC,
        DR,
        DW
    );
    for &(off, len) in &[(lay.l0, DC), (lay.bc, DC), (lay.uvb, DR), (lay.vb, DW), (lay.gb, DW), (lay.bo, NOUT)] {
        for i in 0..len {
            net.p[off + i] += 0.1 * rng.normal() as f64;
        }
    }
    let samples: Vec<Sample> = (0..6).map(|_| sample_world(&mut rng, &pr, 100_000).expect("monde")).collect();
    let mut ws = Work::new();
    let mut z = [0f32; Z_DIM];
    encode(&samples[0].prog, &samples[0].view, &mut z);
    net.forward(&z, 16, &mut ws);
    let long: Vec<f64> = ws.y[..8 * NOUT].to_vec();
    net.forward(&z, 8, &mut ws);
    let causal = ws.y[..8 * NOUT] == long[..];
    println!("causalité (y_k indépendant de T et de n) : {}", if causal { "ok" } else { "ÉCHEC" });
    let total = |n: &Net, w: &mut Work| -> f64 {
        samples.iter().map(|s| n.run_sample(s, 1.0, w, None).loss).sum::<f64>()
    };
    let mut ga = vec![0.0f64; net.p.len()];
    for s in &samples {
        net.run_sample(s, 1.0, &mut ws, Some(&mut ga[..]));
    }
    let eps = 1e-5;
    let mut ok = causal;
    println!("{:<4} {:>10} {:>10} {:>10}", "grp", "|g| max", "err abs", "err rel*");
    for (name, off, len) in lay.groups() {
        let mut idx: Vec<usize> = (0..len).collect();
        idx.sort_by(|&a, &b| ga[off + b].abs().total_cmp(&ga[off + a].abs()));
        let gmax = ga[off + idx[0]].abs();
        let nt = 8.min(len);
        let mut probes: Vec<usize> = idx[..nt].to_vec();
        for _ in 0..8 {
            probes.push(rng.below(len));
        }
        let (mut eabs, mut erel) = (0.0f64, 0.0f64);
        for (q, &i) in probes.iter().enumerate() {
            let j = off + i;
            let x0 = net.p[j];
            net.p[j] = x0 + eps;
            let lp = total(&net, &mut ws);
            net.p[j] = x0 - eps;
            let lm = total(&net, &mut ws);
            net.p[j] = x0;
            let gn = (lp - lm) / (2.0 * eps);
            let d = (ga[j] - gn).abs();
            eabs = eabs.max(d);
            if q < nt {
                erel = erel.max(d / ga[j].abs().max(gn.abs()).max(1e-300));
            }
        }
        let good = gmax > 0.0 && erel < 1e-4 && eabs < 1e-6;
        ok &= good;
        println!("{:<4} {:>10.3e} {:>10.2e} {:>10.2e}  {}", name, gmax, eabs, erel, if good { "ok" } else { "ÉCHEC" });
    }
    println!("(* erreur relative sur les 8 plus grands gradients du groupe ; seuils : rel < 1e-4, abs < 1e-6)");
    println!("{}", if ok { "GRADCHECK OK" } else { "GRADCHECK ÉCHEC" });
    ok
}

pub fn stages(fin: &Prior) -> Vec<Prior> {
    let t0 = 6usize.max(fin.t_lo);
    let q0 = 3usize.max(fin.q_lo);
    let nt = (fin.t_hi.saturating_sub(t0) + 1) / 2;
    let nq = fin.q_hi.saturating_sub(q0);
    let n = nt.max(nq) + 1;
    (0..n)
        .map(|i| {
            let mut p = fin.clone();
            p.t_hi = (t0 + 2 * i).min(fin.t_hi);
            p.q_hi = (q0 + i).min(fin.q_hi);
            p
        })
        .collect()
}

    let par = draw_world(rng, pr, Part::Train, 1_000_000).expect("prior d'entraînement vide");
    if rng.bernoulli(mut_frac) {
        if let Some(c) = mutate_sample(rng, &par, pr, 200) {
            if in_part(&c.prog, Part::Train) {
                return c;
            }
        }
    }
    par
}
struct Worker {
    rng: Rng,
    ws: Work,
    g: Vec<f64>,
}
impl Worker {
    fn new(rng: Rng, np: usize) -> Worker {
        Worker { rng, ws: Work::new(), g: vec![0.0; np] }
    }
    fn run(&mut self, net: &Net, pr: &Prior, n: usize, mut_frac: f64, w_trace: f64) -> Agg {
        for x in self.g.iter_mut() {
            *x = 0.0;
        }
        self.ws.ticks = 0;
        self.ws.capped = 0;
        let mut a = Agg::default();
        for _ in 0..n {
            let s = draw_train(&mut self.rng, pr, mut_frac);
            let st = net.run_sample(&s, w_trace, &mut self.ws, Some(&mut self.g[..]));
            if nontrivial(&s) {
                a.add(&s, &st);
            }
        }
        a
    }
}
pub fn train(cfg: &Cfg) -> (Net, Prior) {
    let mut fin = Prior::demo();
    fin.k_max = cfg.kmax;
    fin.trunc = cfg.trunc;
    let sts: Vec<Prior> = match cfg.fixed {
        Some((t, q)) => {
            let mut p = fin.clone();
            p.t_hi = t.max(p.t_lo);
            p.q_hi = q.max(p.q_lo);
            vec![p]
        }
        None => stages(&fin),
    };
    let nst = sts.len();
    let mut rng = Rng::new(cfg.seed);
    let mut net = Net::new(&mut rng);
    if !cfg.init.is_empty() {
        match Net::load(&cfg.init) {
            Ok(n) => {
                net = n;
                println!("poids initiaux : {}", cfg.init);
            }
            Err(e) => {
                eprintln!("--init {} : {}", cfg.init, e);
                std::process::exit(1);
            }
        }
    }
    net.rho = cfg.rho;
    let np = net.p.len();
    let nth = cfg.threads.clamp(1, cfg.batch.max(1));
    let mut workers: Vec<Worker> = (0..nth).map(|_| Worker::new(rng.fork(), np)).collect();
    let mut vrng = rng.fork();
    let mut vws = Work::new();
    let mut adam = Adam::new(np);
    let mut g = vec![0.0f64; np];
    let groups = net.lay.groups();
    let mut gq = vec![0.0f64; groups.len()];
    let (mut cap_t, mut cap_c) = (0usize, 0usize);
    let cycle = cfg.cycle.max(1);
    let every = cfg.every.max(1);
    let mut stage = cfg.from.min(nst - 1);
    let stage0 = stage;
    println!(
        "entraînement de D (v4) : {} paramètres, {} fil(s), lot {}, {} étage(s){}, départ étage {}, k ≤ {}, mutés {:.2}",
        np,
        nth,
        cfg.batch,
        nst,
        if cfg.fixed.is_some() { " (plage fixe)" } else { "" },
        stage + 1,
        fin.k_max,
        cfg.mut_frac
    );
    println!(
        "lr {:.1e} (étages suivants {:.1e}, montée {} pas) → {:.1e} par cycle de {} pas (crête ×{:.2} par cycle de plus), ≤ {} cycles/étage ; plafond de gain rétrograde ρ = {} ; portes : up {:.0}% / final {:.0}% (n = {})",
        cfg.lr,
        cfg.lr_stage,
        cfg.warm,
        cfg.lr_min,
        cycle,
        cfg.peak_decay,
        cfg.max_cycles,
        cfg.rho,
        100.0 * cfg.up,
        100.0 * cfg.gate,
        cfg.n_gate
    );
    println!(
        "génération : {}",
        if cfg.trunc { "troncature (horizon coupé au dernier tick dont la file lue est ≤ q)" } else { "rejet par qmax (v1)" }
    );
    println!("étage {}/{} : T ≤ {}, qmax ≤ {}", stage + 1, nst, sts[stage].t_hi, sts[stage].q_hi);
    let clock = Instant::now();
    let (mut cyc, mut k, mut step, mut done) = (0usize, 0usize, 0usize, 0usize);
    let mut win = Agg::default();
    let (mut gsum, mut gcnt, mut bad) = (0.0f64, 0usize, 0usize);
    let mut why = "plafond de pas";
    let mut last_stage = stage;
    while step < cfg.steps {
        let pr = &sts[stage];
        last_stage = stage;
        let first = if stage > stage0 { cfg.lr_stage } else { cfg.lr };
        let peak = (first * cfg.peak_decay.powi(cyc as i32)).max(cfg.lr_min);
        let mut lr = cfg.lr_min
            + 0.5 * (peak - cfg.lr_min) * (1.0 + (std::f64::consts::PI * k as f64 / cycle as f64).cos());
        if stage > stage0 && cyc == 0 && k < cfg.warm {
            lr *= (k + 1) as f64 / cfg.warm as f64;
        }
        let (base, extra) = (cfg.batch / nth, cfg.batch % nth);
        let mut aggs: Vec<Agg> = Vec::with_capacity(nth);
        std::thread::scope(|sc| {
            let mut hs = Vec::with_capacity(nth);
            for (i, w) in workers.iter_mut().enumerate() {
                let n = base + usize::from(i < extra);
                let netr = &net;
                hs.push(sc.spawn(move || w.run(netr, pr, n, cfg.mut_frac, cfg.w_trace)));
            }
            for h in hs {
                aggs.push(h.join().expect("fil d'entraînement"));
            }
        });
        for x in g.iter_mut() {
            *x = 0.0;
        }
        for w in &workers {
            for (a, b) in g.iter_mut().zip(&w.g) {
                *a += *b;
            }
            cap_t += w.ws.ticks;
            cap_c += w.ws.capped;
        }
        let inv = 1.0 / cfg.batch as f64;
        for x in g.iter_mut() {
            *x *= inv;
        }
        for (i, &(_, off, len)) in groups.iter().enumerate() {
            gq[i] += g[off..off + len].iter().map(|x| x * x).sum::<f64>();
        }
        let gn = adam.step(&mut net.p, &g, lr, cfg.clip);
        if gn.is_finite() {
            gsum += gn;
            gcnt += 1;
        } else {
            bad += 1;
        }
        for a in &aggs {
            win.merge(a);
        }
        done += cfg.batch;
        step += 1;
        k += 1;
        let timeout = cfg.secs > 0.0 && clock.elapsed().as_secs_f64() >= cfg.secs;
        let cyc_end = k >= cycle;
        if step % every == 0 || cyc_end || timeout {
            let nv = if cyc_end { cfg.n_gate } else { cfg.n_val };
            let v = validate(&net, pr, nv, &mut vrng, &mut vws);
            let el = clock.elapsed().as_secs_f64();
            let tot_q = gq.iter().sum::<f64>().max(1e-300);
            let mut ord: Vec<usize> = (0..groups.len()).collect();
            ord.sort_by(|&a, &b| gq[b].total_cmp(&gq[a]));
            let top: Vec<String> = ord
                .iter()
                .take(3)
                .map(|&i| format!("{} {:.0}%", groups[i].0, 100.0 * gq[i] / tot_q))
                .collect();
            println!(
                "pas {:>6}  étage {}/{} (T≤{}, q≤{})  cycle {}/{}  lr {:.1e}  L {:.4}  train {:>5.1}%  val {:>5.1}% ±{:.1}{}  |g| {:.3}  plaf {:.0}%  énergie {}  {:.0} éch/s  {:.0} s",
                step,
                stage + 1,
                nst,
                pr.t_hi,
                pr.q_hi,
                cyc + 1,
                cfg.max_cycles,
                lr,
                win.mean_loss(),
                100.0 * win.fid(),
                100.0 * v.fid(),
                100.0 * v.ci(),
                if cyc_end { " [porte]" } else { "" },
                gsum / gcnt.max(1) as f64,
                100.0 * cap_c as f64 / cap_t.max(1) as f64,
                top.join(" "),
                done as f64 / el.max(1e-9),
                el
            );
            if bad > 0 {
                println!("  ! {} pas ignorés (gradient non fini)", bad);
            }
            win = Agg::default();
            gsum = 0.0;
            gcnt = 0;
            bad = 0;
            cap_t = 0;
            cap_c = 0;
            for x in gq.iter_mut() {
                *x = 0.0;
            }
            checkpoint(&net, &cfg.out, pr);
            if cyc_end {
                let is_final = stage + 1 == nst;
                let thr = if is_final { cfg.gate } else { cfg.up };
                if v.fid() >= thr {
                    if is_final {
                        why = "étage final maîtrisé";
                        break;
                    }
                    let zs = validate(&net, &sts[stage + 1], cfg.n_gate, &mut vrng, &mut vws);
                    println!(
                        "  ↑ porte franchie ({:.1}% ≥ {:.0}%) → étage {}/{} : T ≤ {}, qmax ≤ {}",
                        100.0 * v.fid(),
                        100.0 * thr,
                        stage + 2,
                        nst,
                        sts[stage + 1].t_hi,
                        sts[stage + 1].q_hi
                    );
                    println!(
                        "    zéro-coup (poids de fin d'étage sur la plage suivante) : fid {:.1}% ±{:.1} (n = {})",
                        100.0 * zs.fid(),
                        100.0 * zs.ci(),
                        zs.n
                    );
                    stage += 1;
                    cyc = 0;
                } else {
                    cyc += 1;
                    println!(
                        "  ✗ porte close ({:.1}% < {:.0}%) : {} cycle(s) sur {} dans l'étage {}/{}",
                        100.0 * v.fid(),
                        100.0 * thr,
                        cyc,
                        cfg.max_cycles,
                        stage + 1,
                        nst
                    );
                    if cyc >= cfg.max_cycles {
                        why = if cfg.fixed.is_some() {
                            "plan fixe terminé (cycles épuisés)"
                        } else {
                            "étage BLOQUÉ (porte jamais franchie)"
                        };
                        break;
                    }
                }
                k = 0;
            }
            if timeout {
                why = "plafond de temps";
                break;
            }
        }
    }
    checkpoint(&net, &cfg.out, &sts[last_stage]);
    println!(
        "fin : {} — dernier étage entraîné {}/{} ; {} pas, {} échantillons, {:.0} s",
        why,
        last_stage + 1,
        nst,
        step,
        done,
        clock.elapsed().as_secs_f64()
    );
    let p = sts[last_stage].clone();
    (net, p)
}

use std::ops::Range;
use std::path::Path;
use chronicle::kv::{Kv, KvEvent};
use chronicle::{Engine, Options, StateMachine};
fn event(i: u32) -> KvEvent {
    let key = format!("k{}", i % 13).into_bytes();
    if i % 5 == 4 {
        KvEvent::Delete { key }
    } else {
        KvEvent::Put { key, value: format!("v{i}").into_bytes() }
    }
}
fn snap(every: u64) -> Options {
    Options { snapshot_every: every, keep_recent: 1000, milestone_every: 0, snapshot_dir: None }
}
fn build(path: &Path, options: Options, range: Range<u32>, batch: u32) -> Vec<Kv> {
    let (mut db, _) = Engine::<Kv>::open_with(path, options).unwrap();
    let mut history = vec![Kv::genesis()];
    for i in range {
        db.submit(&event(i)).unwrap();
        history.push(db.state().clone());
        if (i + 1) % batch == 0 {
            db.commit().unwrap();
        }
    }
    db.commit().unwrap();
    history
}

fn time_travel_and_cursor_match_history() {
    let dir = tempfile::tempdir().unwrap();
    let path = dir.path().join("kv.log");
    let history = build(&path, snap(40), 0..300, 20);
    let (db, _) = Engine::<Kv>::open_with(&path, snap(40)).unwrap();
    assert!(db.boot_seq() > 0);
    for (cycle, expected) in history.iter().enumerate() {
        assert_eq!(&db.state_at(cycle as u64).unwrap(), expected, "state_at({cycle})");
    }
    let mut cur = db.cursor_at(0).unwrap();
    assert_eq!(cur.peek().unwrap(), Some(event(0)));
    while cur.step_forward().unwrap().is_some() {
        assert_eq!(cur.state(), &history[cur.seq() as usize]);
    }
    assert!(cur.at_end());
    while let Some(undone) = cur.step_back().unwrap() {
        assert_eq!(cur.state(), &history[cur.seq() as usize]);
        assert_eq!(undone, event(cur.seq() as u32)); 
    }
    assert_eq!(cur.seq(), 0);
    cur.goto(250).unwrap();
    assert_eq!(cur.state(), &history[250]);
    cur.goto(123).unwrap();
    assert_eq!(cur.state(), &history[123]);
    assert!(cur.goto(301).is_err());
    cur.goto(0).unwrap();
    let first = history.iter().position(|s| s.len() >= 8).map(|c| c as u64);
    assert!(first.is_some());
    assert_eq!(cur.run_until(|s, _| s.len() >= 8).unwrap(), first);
}
